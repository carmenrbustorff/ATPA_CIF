import os

ARCHITECTURE_DEPTH_OPTIONS = {0: 2, 1: 3, 2: 4}
DROPOUT_RATES = [0.0, 0.2, 0.5]
REGULARIZATION_L2 = [0, 0.0001, 0.001]

batch_size = BATCH_SIZE_OPTIONS[ITERATION % 3]

num_conv_layers = ARCHITECTURE_DEPTH_OPTIONS[ITERATION % 3]

l2_reg = REGULARIZATION_L2[ITERATION // 3]

train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=3, dropout_rate=0.2):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        out_channels = 32
        
        # Add convolutional layers based on the specified depth
        for i in range(num_conv_layers):
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
            if i == 0:
                out_channels *= 2
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.flatten = nn.Flatten()
        self.classifier = nn.Sequential(
            nn.Linear(out_channels, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        x = self.flatten(x)
        return torch.sigmoid(self.classifier(x))
