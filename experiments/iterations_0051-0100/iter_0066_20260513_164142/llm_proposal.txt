```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants for iteration 15
ITERATION = 15
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Hyperparameter selection based on iteration number
if ITERATION % 3 == 0:
    batch_size = 8
    lr = 0.001
    num_conv_layers = 2
elif ITERATION % 3 == 1:
    batch_size = 16
    lr = 0.0001
    num_conv_layers = 3
else:
    batch_size = 32
    lr = 0.00001
    num_conv_layers = 4

# Dropout and L2 regularization options (example values)
dropout_rate = 0.5 if ITERATION % 3 == 2 else 0.2 if ITERATION % 3 == 1 else 0.0
weight_decay = 0.001 if ITERATION % 3 == 2 else 0.0001 if ITERATION % 3 == 1 else 0

# Load the dataset
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

class BirdClassifier(nn.Module):
    def __init__(self, num_conv_layers=3, dropout_rate=0.5):
        super(BirdClassifier, self).__init__()
        layers = []
        in_channels = 1
        for i in range(num_conv_layers):
            out_channels = 32 * (i + 1)
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.flatten = nn.Flatten()
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, 234)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, optimizer, and loss function
model = BirdClassifier(num_conv_layers=num_conv_layers, dropout_rate=dropout_rate).to('cuda')
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)
criterion = nn.BCELoss()

# Training loop
epochs_trained = 2
final_train_loss = None

for epoch in range(epochs_trained):
    model.train()
    running_loss = 0.0
    batch_count = 0
    for inputs, labels in train_loader:
        if batch_count >= 10:  # Limit to N batches for fast feedback
            break
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        torch.cuda.empty_cache()  # Clear GPU memory
        batch_count += 1
    
    final_train_loss = running_loss / batch_count
    print(f"Epoch {epoch+1}, Loss: {final_train_loss}")

# Evaluation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        if batch_count >= 20:  # Limit to N batches for fast feedback (separate from training)
            break
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Metrics calculation and saving
num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
training_samples = batch_count * batch_size
eval_samples = len(all_labels)

metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": epochs_trained,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": eval_samples
}

print("METRICS: " + json.dumps(metrics))

# Save model and metrics
torch.save(model.state_dict(), 'model.pt')
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
```

### Explanation:
This code defines a BirdClassifier with a flexible number of convolutional layers based on the iteration number. It uses adaptive pooling to ensure robust flattening and avoids BatchNorm1d in the classifier as per the rules. The training loop is limited to 2 epochs and 10 batches for fast feedback, following the constraints. Evaluation is performed using the same DataLoader in evaluation mode, and metrics such as AUC are computed and saved to a JSON file along with the model weights. This setup allows for rapid experimentation by varying hyperparameters systematically based on iteration number.
