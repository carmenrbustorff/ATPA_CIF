```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define hyperparameters
NUM_SPECIES = 206
ITERATION = 3  # Change this to the current iteration number for systematic exploration
BATCH_SIZE_OPTIONS = [8, 16, 32]
LEARNING_RATE_OPTIONS = [0.001, 0.0001, 0.00001]
CONV_LAYERS_OPTIONS = [2, 3, 4]
DROPOUT_OPTIONS = [0.0, 0.2, 0.5]

# Select hyperparameters based on iteration number
batch_size = BATCH_SIZE_OPTIONS[ITERATION % 3]
lr = LEARNING_RATE_OPTIONS[ITERATION % 3]
num_conv_layers = CONV_LAYERS_OPTIONS[ITERATION % 3]
dropout_rate = DROPOUT_OPTIONS[ITERATION % 3]

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Subset the dataset for faster iteration
training_samples = 15000
subset_indices = list(range(min(training_samples, len(train_loader.dataset))))
train_subset = torch.utils.data.Subset(train_loader.dataset, subset_indices)
train_loader = torch.utils.data.DataLoader(train_subset, batch_size=batch_size, shuffle=True)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, num_conv_layers, dropout_rate):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential()
        in_channels = 1
        
        for i in range(num_conv_layers):
            out_channels = 32 * (i + 1)
            self.conv_layers.add_module(f'conv{i+1}', nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            self.conv_layers.add_module(f'bn{i+1}', nn.BatchNorm2d(out_channels))
            self.conv_layers.add_module(f'relu{i+1}', nn.ReLU())
            in_channels = out_channels
        
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.flatten = nn.Flatten()
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Instantiate model
model = BirdCLEFModel(NUM_SPECIES, num_conv_layers, dropout_rate)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Move model to GPU if available
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Training loop
epochs = 6
for epoch in range(epochs):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        torch.cuda.empty_cache()  # Clear GPU cache
    
    print(f'Epoch [{epoch+1}/{epochs}], Loss: {running_loss/len(train_loader):.4f}')

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Calculate AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f'Final AUC: {final_auc:.4f}')

# Save metrics and model
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": len(all_labels)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print(f'METRICS: {json.dumps(metrics)}')
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:
1. **Imports and Hyperparameters**: The script starts with the required imports and defines hyperparameters based on the iteration number for systematic exploration of different configurations.
2. **Data Loading**: Uses the pre-built PyTorch DataLoader to load data, ensuring compliance with competition rules.
3. **Model Architecture**: Defines a CNN model with a specified number of convolutional layers, batch normalization, ReLU activation, and dropout for regularization.
4. **Training Loop**: Includes moving the model to GPU if available, training for 6 epochs, and clearing GPU cache after each batch to prevent memory fragmentation.
5. **Evaluation**: Evaluates the model on the same DataLoader in eval mode, calculates AUC using `roc_auc_score`, and saves metrics and model state.
6. **Metrics and Model Saving**: Saves computed metrics to a JSON file and the model's state dictionary for future use.

This approach ensures rapid experimentation with different hyperparameters while adhering to the competition rules and constraints.
