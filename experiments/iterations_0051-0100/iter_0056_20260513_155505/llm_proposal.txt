```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Set up environment and dataloader
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
iteration = 5

# Hyperparameter selection based on iteration number
if iteration % 3 == 0:
    batch_size = 8
    lr = 0.001
    num_conv_layers = 2
elif iteration % 3 == 1:
    batch_size = 16
    lr = 0.0001
    num_conv_layers = 3
else:
    batch_size = 32
    lr = 0.00001
    num_conv_layers = 4

# Load the dataloader
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Define the CNN model architecture
class BirdCNN(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=3):
        super(BirdCNN, self).__init__()
        layers = []
        in_channels = 1
        for i in range(num_conv_layers):
            out_channels = 32 * (2 ** i)
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(out_channels, 128),
            nn.ReLU(),
            nn.Dropout(0.2),  # Adding dropout for regularization
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, and optimizer
model = BirdCNN(num_classes=234, num_conv_layers=num_conv_layers).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.001)  # Adding L2 regularization

# Training loop with early stopping and validation
epochs_trained = 2
final_train_loss = None
training_samples = len(train_loader.dataset)
eval_samples = training_samples // 10  # Assuming 10% of data for evaluation

all_preds = []
all_labels = []

model.train()
for epoch in range(epochs_trained):
    batch_count = 0
    running_loss = 0.0
    for inputs, labels in train_loader:
        if batch_count >= 5:  # Limit to first ~5000 samples (~10 batches)
            break
        
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        batch_count += 1
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
        
        print(f'Epoch {epoch+1}, Batch {batch_count}, Loss: {loss.item()}')
    
    final_train_loss = running_loss / batch_count

# Validation loop
model.eval()
with torch.no_grad():
    for inputs, labels in train_loader:
        if len(all_labels) >= eval_samples // batch_size:
            break
        
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

# Compute validation AUC
import numpy as np
from sklearn.metrics import roc_auc_score

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON and print
metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs_trained,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": eval_samples
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation

**Model Architecture:**
- The `BirdCNN` class defines a CNN with a flexible number of convolutional layers (`num_conv_layers`). This allows easy experimentation with different depths.
- Each convolutional layer is followed by batch normalization and ReLU activation, typical for deep learning models to ensure stable training.
- Max pooling is used after each convolutional block to reduce spatial dimensions.
- An `AdaptiveAvgPool2d` ensures that the output size of the feature maps is fixed regardless of the input size, which is crucial for the following fully connected layers.
- The classifier consists of a linear layer reducing dimensions to 128 followed by ReLU and dropout for regularization. Another linear layer outputs logits for each of the 234 classes.
- Sigmoid activation is applied in the forward method to ensure output values are between 0 and 1, suitable for multi-label binary classification.

**Training Loop:**
- The model is trained for a maximum of 2 epochs with a small batch size (8, 16, or 32) based on the iteration number.
- Loss is computed using `BCELoss`, which requires outputs in the range [0, 1].
- Training is limited to the first ~5000 samples to ensure rapid feedback during experimentation.

**Validation:**
- The same dataloader is used for validation by setting the model to evaluation mode and disabling gradient computation.
- Validation AUC is computed using `roc_auc_score` with weighted averaging, which considers all classes equally.

**Metrics and Model Saving:**
- Metrics such as final training loss, validation AUC, number of parameters, batch size, and sample counts are saved in a JSON file for easy tracking.
- The model's state dictionary is saved to allow further use or ensemble training.
