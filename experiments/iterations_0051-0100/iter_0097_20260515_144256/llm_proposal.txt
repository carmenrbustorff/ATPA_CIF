```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Define hyperparameters based on the iteration number (e.g., Iteration 1)
iteration = 1
batch_size_options = [8, 16, 32]
lr_options = [0.001, 0.0001, 0.00001]
conv_depth_options = [2, 3, 4]

batch_size = batch_size_options[iteration % 3]
lr = lr_options[iteration % 3]
conv_depth = conv_depth_options[iteration % 3]
dropout_rate = 0.5 if iteration == 3 else (0.2 if iteration == 1 else 0.0)
num_species = 206

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Model definition
class BirdClassifier(nn.Module):
    def __init__(self, num_species, conv_depth, dropout_rate):
        super(BirdClassifier, self).__init__()
        self.conv_layers = nn.Sequential()
        for i in range(conv_depth):
            in_channels = 32 if i > 0 else 1
            out_channels = 32 * (i + 1)
            self.conv_layers.add_module(f'conv_{i}', nn.Conv2d(in_channels, out_channels, 3, padding=1))
            self.conv_layers.add_module(f'bn_{i}', nn.BatchNorm2d(out_channels))
            self.conv_layers.add_module(f'relu_{i}', nn.ReLU())
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(32 * conv_depth, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdClassifier(num_species, conv_depth, dropout_rate)
if torch.cuda.is_available():
    model.to('cuda')

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Training loop
epochs = 6
train_loss_history = []
for epoch in range(epochs):
    model.train()
    for inputs, labels in tqdm(train_loader):
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_loss_history.append(loss.item())
        torch.cuda.empty_cache()

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(train_loader):
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

all_labels = np.concatenate(all_labels)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics
metrics = {
    "final_train_loss": np.mean(train_loss_history),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(train_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print(f"METRICS: {json.dumps(metrics)}")

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: The model uses a sequence of convolutional layers followed by batch normalization and ReLU activation. The depth of the convolutional layers is determined by `conv_depth`, which varies based on the iteration number.

2. **Training Loop**: The training loop runs for 6 epochs, with loss reported per batch. The model is moved to CUDA if available, and gradients are zeroed before each forward-backward pass.

3. **Evaluation**: After training, the model is evaluated using the same `train_loader` in eval mode. The AUC score is computed using `roc_auc_score`.

4. **Metrics and Model Saving**: The final training loss, AUC score, number of parameters, and other relevant metrics are saved to `metrics.json`. The model's state dictionary is saved as `model.pt`.

This approach ensures rapid experimentation by systematically varying hyperparameters and allowing for quick iterations.
