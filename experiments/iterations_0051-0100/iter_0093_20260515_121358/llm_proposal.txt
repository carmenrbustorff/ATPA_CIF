```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_SPECIES = 206

# Hyperparameters based on iteration number (e.g., Iteration 4)
ITERATION = 4
BATCH_SIZE = 16 if ITERATION % 3 == 1 else (8 if ITERATION % 3 == 0 else 32)
LEARNING_RATE = 0.0001 if ITERATION % 3 == 1 else (0.001 if ITERATION % 3 == 0 else 0.00001)
CONV_LAYERS = 3 if ITERATION % 3 == 1 else (2 if ITERATION % 3 == 0 else 4)
DROPOUT_RATE = 0.3
L2_REGULARIZATION = 0.0001

# Data loaders
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=BATCH_SIZE)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, conv_layers=CONV_LAYERS, dropout_rate=DROPOUT_RATE):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for i in range(conv_layers):
            out_channels = 32 * (i + 1)
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Instantiate model, loss function, optimizer
model = BirdCLEFModel(num_species=NUM_SPECIES, conv_layers=CONV_LAYERS, dropout_rate=DROPOUT_RATE)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=L2_REGULARIZATION)

# Move model to CUDA
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)

# Training loop
epochs = 6
train_loss_history = []
for epoch in range(epochs):
    model.train()
    total_loss = 0.0
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item()
        torch.cuda.empty_cache()  # Clear GPU cache
        
    avg_train_loss = total_loss / len(train_loader)
    train_loss_history.append(avg_train_loss)
    print(f"Epoch {epoch+1}, Train Loss: {avg_train_loss:.4f}")

# Evaluation on a subset of the validation set
val_subset = torch.utils.data.Subset(train_loader.dataset, range(min(5000, len(train_loader.dataset))))
val_loader = torch.utils.data.DataLoader(val_subset, batch_size=32, shuffle=False)

model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

# Flatten predictions and labels
import numpy as np
all_preds_flat = np.concatenate(all_preds)
all_labels_flat = np.concatenate(all_labels)

# Calculate AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels_flat, all_preds_flat, average='weighted')
print(f"Final AUC: {final_auc:.4f}")

# Save metrics to JSON
metrics = {
    "final_train_loss": train_loss_history[-1],
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_subset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:
- **Model Architecture**: The model uses a series of convolutional layers followed by batch normalization and ReLU activation. It then applies `AdaptiveAvgPool2d` to handle different input sizes robustly, flattens the output, and passes it through fully connected layers with dropout for regularization.
- **Training Loop**: The loop iterates over the training data, computes loss, and updates weights using Adam optimizer. It includes a call to `torch.cuda.empty_cache()` after each batch to manage GPU memory efficiently.
- **Evaluation**: The model is evaluated on a subset of the dataset to compute AUC using `roc_auc_score`.
- **Metrics & Checkpointing**: Training metrics are saved in JSON format, and the model's state dictionary is saved for later use.
