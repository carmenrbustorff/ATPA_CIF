```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants for the current iteration
ITERATION = 19
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Hyperparameter selection based on iteration number
batch_size_options = [8, 16, 32]
lr_options = [0.001, 0.0001, 0.00001]
conv_depth_options = [2, 3, 4]

batch_size = batch_size_options[ITERATION % 3]
lr = lr_options[ITERATION % 3]
conv_depth = conv_depth_options[ITERATION % 3]
dropout_rate = 0.2

# Data loading
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, conv_depth=3, dropout_rate=0.2):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for i in range(conv_depth):
            out_channels = 32 * (2 ** i)
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.dropout = nn.Dropout(dropout_rate)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Linear(in_channels, num_classes)

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        x = self.dropout(x)
        return torch.sigmoid(self.classifier(x))

# Model setup
model = BirdCLEFModel(conv_depth=conv_depth, dropout_rate=dropout_rate).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Training loop
num_epochs = 2
training_samples = 0
final_train_loss = 0.0

model.train()
for epoch in range(num_epochs):
    batch_count = 0
    for inputs, labels in train_loader:
        if training_samples >= 5000:  # Limit to ~5000 samples for fast feedback
            break
        
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        final_train_loss += loss.item()
        training_samples += len(inputs)
        batch_count += 1
        
        print(f'Epoch [{epoch+1}/{num_epochs}], Batch [{batch_count}], Loss: {loss.item():.4f}')
        
        torch.cuda.empty_cache()

# Evaluation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        if training_samples >= 5000:  # Limit to ~5000 samples for fast feedback
            break
        
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Metrics and model saving
num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
metrics = {
    "final_train_loss": final_train_loss / batch_count,
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": training_samples
}

print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
```

### Explanation

- **Model Architecture**: The model is a CNN with a variable number of convolutional layers (2, 3, or 4 based on the iteration number). Each convolutional layer is followed by BatchNorm and ReLU activation. Max pooling reduces dimensions after each conv block. Dropout is applied before the final linear layer to prevent overfitting.
  
- **Training**: The model trains for 2 epochs using BCELoss with a sigmoid activation in the forward pass to ensure outputs are between [0,1]. Hyperparameters like batch size and learning rate vary based on the iteration number to explore different configurations rapidly.

- **Evaluation**: After training, the model is evaluated on the same dataset (in evaluation mode) by computing the weighted AUC score across all 234 classes.

- **Metrics and Saving**: The script saves metrics including training loss, validation AUC, number of parameters, and other relevant information to a JSON file. Model weights are also saved for potential reuse or further analysis.
