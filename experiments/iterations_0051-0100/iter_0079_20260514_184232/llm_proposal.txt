```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Data ingestion
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
batch_size = 16
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Define the model architecture based on iteration rules
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdCLEFModel, self).__init__()
        # Convolutional layers
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
        )
        # Adaptive pooling
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        # Classifier layers
        self.classifier = nn.Sequential(
            nn.Linear(128, 256), nn.ReLU(), nn.Dropout(0.3),
            nn.Linear(256, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, and optimizer
model = BirdCLEFModel()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.0001)

# Move model to CUDA if available
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)

# Training loop
num_epochs = 2
total_train_samples = 5000
train_losses = []

for epoch in range(num_epochs):
    model.train()
    batch_count = 0
    for inputs, labels in train_loader:
        if batch_count >= total_train_samples // batch_size:
            break
        
        # Move data to device
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        
        # Backward pass and optimization
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        # Report loss per batch
        train_losses.append(loss.item())
        print(f"Epoch [{epoch+1}/{num_epochs}], Batch [{batch_count+1}/{total_train_samples // batch_size}], Loss: {loss.item():.4f}")
        
        # Clear cache to prevent GPU fragmentation
        torch.cuda.empty_cache()
        
        batch_count += 1

# Evaluation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

# Flatten predictions and labels
import numpy as np
all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

# Calculate AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON
metrics = {
    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": total_train_samples,
    "eval_samples": len(all_labels)
}
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: The `BirdCLEFModel` class defines a CNN with two convolutional layers followed by batch normalization and ReLU activation. An adaptive average pooling layer is used to flatten the features, and a classifier with two linear layers (and dropout for regularization) outputs the final probabilities.

2. **Training Loop**: The model is trained for 2 epochs using Adam optimizer and BCELoss. The training loop limits itself to approximately 5000 samples (10 batches).

3. **Evaluation**: The model is evaluated on the same DataLoader in eval mode, and AUC is computed using `roc_auc_score`.

4. **Metrics and Checkpoint**: Training loss, AUC, number of parameters, and other relevant metrics are saved to a JSON file, and the model's state dictionary is saved for later use.

This architecture and training loop are designed for rapid experimentation and iteration, allowing for quick feedback on different hyperparameters and configurations.
