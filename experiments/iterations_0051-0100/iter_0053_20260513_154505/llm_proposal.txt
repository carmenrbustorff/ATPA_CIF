```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Hyperparameters based on iteration number (Iteration 2)
iteration = 2
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

batch_size_options = {0: 8, 1: 16, 2: 32}
learning_rate_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
conv_layer_count_options = {0: 2, 1: 3, 2: 4}

batch_size = batch_size_options[iteration % 3]
lr = learning_rate_options[iteration % 3]
num_conv_layers = conv_layer_count_options[iteration % 3]

# Set up DataLoader
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=2):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for i in range(num_conv_layers):
            out_channels = 32 * (2 ** i)
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(out_channels, 128),
            nn.ReLU(),
            nn.Dropout(0.5),  # Adding dropout for regularization
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        x = self.classifier(x)
        return torch.sigmoid(x)

# Initialize model, loss function, and optimizer
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = BirdCLEFModel(num_conv_layers=num_conv_layers).to(device)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.001)  # L2 regularization

# Training loop
num_epochs = 2
batch_count_limit = 10  # Limit to ~5000 samples for fast feedback
train_losses = []

model.train()
for epoch in range(num_epochs):
    batch_count = 0
    for inputs, labels in tqdm(train_loader):
        if batch_count >= batch_count_limit:
            break
        
        optimizer.zero_grad()
        
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        print(f"Epoch [{epoch+1}/{num_epochs}], Batch [{batch_count+1}/{batch_count_limit}], Loss: {loss.item():.4f}")
        
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
        batch_count += 1

# Evaluation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Validation AUC: {final_auc:.4f}")

# Calculate number of parameters
num_params = sum(p.numel() for p in model.parameters())

# Metrics and saving
metrics = {
    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": batch_count_limit * batch_size,
    "eval_samples": batch_count_limit * batch_size
}

print("METRICS: " + json.dumps(metrics))
torch.save(model.state_dict(), 'model.pt')
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
```

### Explanation:

1. **Model Architecture**: The `BirdCLEFModel` class is designed to have a flexible number of convolutional layers based on the iteration number. This allows for easy experimentation with model depth. Each convolutional layer is followed by batch normalization and ReLU activation, with max pooling applied after each set of convolutions.

2. **Regularization**: Dropout is added in the classifier section to help prevent overfitting, especially given the limited amount of data used during training.

3. **Training Loop**: The model is trained for 2 epochs, iterating over a subset of the dataset (limited by `batch_count_limit`). Loss is printed per batch, and GPU memory fragmentation is mitigated using `torch.cuda.empty_cache()` after each batch.

4. **Evaluation Loop**: After training, the model evaluates on the same dataset in evaluation mode. Predictions are collected and used to compute the weighted AUC score.

5. **Metrics & Saving**: The script calculates and prints metrics such as final train loss, validation AUC, number of parameters, etc. These metrics are saved to a JSON file (`metrics.json`), and the model's state dictionary is saved to `model.pt` for later use.

This setup ensures rapid iteration by limiting the dataset size and epochs while still providing meaningful feedback on model performance.
