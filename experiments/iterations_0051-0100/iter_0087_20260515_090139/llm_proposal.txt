```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Hyperparameters
NUM_CLASSES = 206
EPOCHS = 6
ITERATION = 1  # Change this for different iterations
BATCH_SIZE = [8, 16, 32][ITERATION % 3]
LR = [0.001, 0.0001, 0.00001][ITERATION % 3]
NUM_CONV_LAYERS = [2, 3, 4][ITERATION % 3]
DROPOUT_RATE = [0.0, 0.2, 0.5][ITERATION // 3]  # Vary dropout rate

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=BATCH_SIZE)

# Model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=NUM_CLASSES, dropout_rate=DROPOUT_RATE, num_conv_layers=NUM_CONV_LAYERS):
        super(BirdCLEFModel, self).__init__()
        conv_layers = []
        in_channels = 1
        
        for i in range(num_conv_layers):
            out_channels = 32 * (i + 1)
            conv_layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            conv_layers.append(nn.BatchNorm2d(out_channels))
            conv_layers.append(nn.ReLU())
            in_channels = out_channels
        
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Dropout(dropout_rate),
            nn.Linear(in_channels, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_classes=NUM_CLASSES, dropout_rate=DROPOUT_RATE, num_conv_layers=NUM_CONV_LAYERS)
if torch.cuda.is_available():
    model.to('cuda')

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LR)

# Training loop
train_losses = []
for epoch in range(EPOCHS):
    model.train()
    batch_count = 0
    for inputs, labels in train_loader:
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        print(f"Epoch {epoch+1}/{EPOCHS}, Batch {batch_count+1}, Loss: {loss.item()}")
        
        torch.cuda.empty_cache()
        batch_count += 1
        if batch_count >= 20:  # Limit to 20 batches for fast feedback
            break

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

import numpy as np
from sklearn.metrics import roc_auc_score

all_preds = np.vstack(all_preds)
all_labels = np.vstack(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Metrics and model saving
metrics = {
    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": 15000,  # Assuming a subset of 15000 samples
    "eval_samples": 5000  # Assuming a subset of 5000 samples for validation
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Imports and Hyperparameters**: The script starts with the required imports and defines hyperparameters based on the iteration number. This includes batch size, learning rate, number of convolutional layers, dropout rate, and other relevant parameters.

2. **Data Loading**: The `get_dataloader` function is used to load the training data. The dataloader returns a single DataLoader object, which is used for both training and validation.

3. **Model Architecture**: A custom CNN model (`BirdCLEFModel`) is defined with the required structure: convolutional layers followed by batch normalization, ReLU activation, adaptive average pooling, flattening, and fully connected layers. The number of convolutional layers and dropout rate are varied based on the iteration number.

4. **Training Loop**: The training loop iterates over a limited number of batches (20) to ensure fast feedback. The model is moved to CUDA if available, and loss and optimizer are defined using BCELoss and Adam respectively. Training losses are stored for later analysis.

5. **Evaluation**: After training, the model is evaluated on the same DataLoader object used for training (limited to 5000 samples). AUC is computed using `roc_auc_score`.

6. **Metrics and Model Saving**: Metrics including final training loss, AUC, number of parameters, epochs trained, batch size, training samples, and evaluation samples are saved to a JSON file. The model's state dictionary is also saved for later use.

This approach allows for rapid experimentation by varying hyperparameters and architecture components based on the iteration number.
