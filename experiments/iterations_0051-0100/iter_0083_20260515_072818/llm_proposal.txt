                nn.ReLU()
            ]
        elif iteration % 3 == 1:
            conv_layers = [
                nn.Conv2d(1, 32, 3, padding=1), 
                nn.BatchNorm2d(32) if use_batch_norm else nn.Identity(), 
                nn.ReLU(),
                nn.Conv2d(32, 64, 3, padding=1), 
                nn.BatchNorm2d(64) if use_batch_norm else nn.Identity(), 
                nn.ReLU(),
                nn.Conv2d(64, 128, 3, padding=1), 
                nn.BatchNorm2d(128) if use_batch_norm else nn.Identity(), 
                nn.ReLU()
            ]
        elif iteration % 3 == 2:
            conv_layers = [
                nn.Conv2d(1, 32, 3, padding=1),

                nn.ReLU(),
                nn.Conv2d(32, 64, 3, padding=1), 
                nn.BatchNorm2d(64) if use_batch_norm else nn.Identity(), 
                nn.ReLU(),
                nn.Conv2d(64, 128, 3, padding=1), 
                nn.BatchNorm2d(128) if use_batch_norm else nn.Identity(), 
                nn.ReLU(),
                nn.Conv2d(128, 256, 3, padding=1), 
                nn.BatchNorm2d(256) if use_batch_norm else nn.Identity(), 
                nn.ReLU()
            ]
        
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Fully connected layers
        classifier_layers = [
            nn.Linear(256 if iteration % 3 == 2 else 128, 128), 
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes)
        ]
        self.classifier = nn.Sequential(*classifier_layers)

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Determine hyperparameters based on iteration number

batch_size_options = {0: 8, 1: 16, 2: 32}
learning_rate_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
architecture_depth_options = {0: 2, 1: 3, 2: 4}
dropout_rates = [0.0, 0.2, 0.5]
use_batch_norm_options = [True, True, False]

lr = learning_rate_options[iteration % 3]
architecture_depth = architecture_depth_options[iteration % 3]

# Load data
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

            break

        optimizer.step()

3. **Training Loop**: The model is trained for a fixed number of epochs (6) using a limited number of training samples (5000) to ensure fast iterations. Losses are tracked during training.

4. **Evaluation**: After training, the model is evaluated on the same dataset used for training to compute the AUC score. This helps in understanding the model's performance without needing a separate validation set.

5. **Metrics and Checkpoint**: Metrics including final training loss, AUC, number of parameters, epochs trained, batch size, and sample sizes are saved to `metrics.json` and printed to stdout. The model's state dictionary is also saved for later use or submission.
