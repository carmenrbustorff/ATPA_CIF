import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Define hyperparameters based on iteration number
iteration = 51  # Example iteration number
batch_size = [8, 16, 32][iteration % 3]
lr = [0.001, 0.0001, 0.00001][iteration % 3]
num_convs = [2, 3, 4][iteration % 3]
dropout_rate = [0.0, 0.2, 0.5][(iteration // 3) % 3]

# Define model architecture
class BirdClassifier(nn.Module):
    def __init__(self, num_species=206):
        super(BirdClassifier, self).__init__()
        conv_layers = []
        in_channels = 1
        for i in range(num_convs):
            out_channels = 32 * (i + 1)
            conv_layers.append(nn.Conv2d(in_channels, out_channels, 3, padding=1))
            conv_layers.append(nn.BatchNorm2d(out_channels))
            conv_layers.append(nn.ReLU())
            in_channels = out_channels
        
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize dataloader
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
dataset = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=32).dataset

# Subset the dataset for faster iteration
train_subset = torch.utils.data.Subset(dataset, range(15000))
val_subset = torch.utils.data.Subset(dataset, range(15000, 20000))

train_loader = torch.utils.data.DataLoader(train_subset, batch_size=batch_size, shuffle=True)
val_loader = torch.utils.data.DataLoader(val_subset, batch_size=batch_size, shuffle=False)

# Initialize model, loss function, and optimizer
model = BirdClassifier(num_species=206).cuda()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Training loop
num_epochs = 6
for epoch in range(num_epochs):
    model.train()
    running_loss = 0.0
    for inputs, labels in tqdm(train_loader):
        optimizer.zero_grad()
        outputs = model(inputs.cuda())
        loss = criterion(outputs, labels.cuda())
        loss.backward()
        optimizer.step()
        running_loss += loss.item()
        
        torch.cuda.empty_cache()  # Clear cache after each batch
    
    print(f"Epoch {epoch+1}, Train Loss: {running_loss/len(train_loader)}")

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(val_loader):
        outputs = model(inputs.cuda())
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Final AUC: {final_auc}")

# Save metrics
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": len(train_subset),
    "eval_samples": len(val_subset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS:", json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
