```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Hyperparameters
epochs = 6
batch_size = 16
lr = 0.0001
num_species = 206

# Data ingestion
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Limit dataset for faster iteration
subset_size = 15000
val_subset_size = 5000

subset = torch.utils.data.Subset(train_loader.dataset, range(min(subset_size, len(train_loader.dataset))))
train_loader.dataset = subset

val_subset = torch.utils.data.Subset(train_loader.dataset, 
                                    range(len(train_loader.dataset) - val_subset_size, 
                                          len(train_loader.dataset)))
val_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)
val_loader.dataset = val_subset

# Model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(128, 256), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(256, num_species)  # NO BatchNorm1d!
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_species=num_species).to('cuda')

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Training loop
train_losses = []
for epoch in range(epochs):
    model.train()
    for inputs, labels in tqdm(train_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        torch.cuda.empty_cache()  # Prevent GPU fragmentation

    print(f"Epoch {epoch+1}, Loss: {sum(train_losses)/len(train_losses)}")

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(val_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics and model checkpoint
metrics = {
    "final_train_loss": sum(train_losses)/len(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": len(subset),
    "eval_samples": len(val_subset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: ", json.dumps(metrics))
torch.save(model.state_dict(), 'model.pt')
```

### Explanation

1. **Imports and Data Ingestion**: We start by importing the required libraries and setting up the data loader using `get_dataloader()`. The dataset is limited to 15,000 samples for training and 5,000 for validation.

2. **Model Architecture**: The model architecture follows a standard CNN with batch normalization layers after each convolutional layer, followed by an adaptive average pooling layer to handle variable input sizes. The classifier consists of fully connected layers with dropout regularization.

3. **Training Loop**: The training loop uses the Adam optimizer and BCELoss for multi-label classification. Loss is reported per epoch, and GPU memory is managed by clearing cache after each batch.

4. **Evaluation**: After training, the model is evaluated on the validation set using the AUC metric. This ensures that the model's performance is assessed in a meaningful way.

5. **Metrics and Model Saving**: Training metrics are saved to `metrics.json`, and the model weights are saved as `model.pt`. This allows for easy reuse and further experimentation.
