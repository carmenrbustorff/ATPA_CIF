```python
import os
import torch
import torch.nn as nn
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define the model architecture with systematic hyperparameter exploration based on iteration number
class BirdClassifier(nn.Module):
    def __init__(self, num_conv_layers=3, dropout_rate=0.2):
        super(BirdClassifier, self).__init__()
        
        conv_layers = []
        in_channels = 1
        out_channels = 32
        
        for _ in range(num_conv_layers):
            conv_layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
            out_channels *= 2
        
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, 234)  # NO BatchNorm1d!
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Set up the DataLoader
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Determine hyperparameters based on iteration number
iteration_number = 17

if iteration_number % 3 == 0:
    batch_size = 8
    lr = 0.001
    num_conv_layers = 2
    dropout_rate = 0.0
elif iteration_number % 3 == 1:
    batch_size = 16
    lr = 0.0001
    num_conv_layers = 3
    dropout_rate = 0.2
else:
    batch_size = 32
    lr = 0.00001
    num_conv_layers = 4
    dropout_rate = 0.5

train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Initialize model, loss function, and optimizer
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = BirdClassifier(num_conv_layers=num_conv_layers, dropout_rate=dropout_rate).to(device)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.001)  # Adding L2 regularization

# Training loop
epochs_trained = 2
final_train_loss = 0.0
batch_count = 0
training_samples = 0

model.train()
for epoch in range(epochs_trained):
    for inputs, labels in train_loader:
        if batch_count >= 10:  # Limit to N batches for fast feedback
            break
        
        # Move data to GPU
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        final_train_loss += loss.item()
        batch_count += 1
        training_samples += inputs.size(0)
        
        # Report loss per batch
        print(f"Epoch [{epoch+1}/{epochs_trained}], Batch [{batch_count}/10], Loss: {loss.item():.4f}")
        
        torch.cuda.empty_cache()  # Prevent GPU fragmentation

# Validation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

import numpy as np
from sklearn.metrics import roc_auc_score

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON
metrics = {
    "final_train_loss": final_train_loss / batch_count,
    "final_auc": float(final_auc),
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs_trained,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": training_samples  # Since we used the same loader for validation
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```
