```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define the CNN model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, dropout_rate=0.0):
        super(BirdCLEFModel, self).__init__()
        
        # Convolutional layers
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.Conv2d(128, 256, kernel_size=3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
            nn.Conv2d(256, 512, kernel_size=3, padding=1), nn.BatchNorm2d(512), nn.ReLU()
        )
        
        # Adaptive pooling layer
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Fully connected layers
        self.classifier = nn.Sequential(
            nn.Linear(512, 256), nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(256, 128), nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
batch_size = 32  # Based on iteration % 3 == 2
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Model setup
model = BirdCLEFModel(num_classes=234, dropout_rate=0.5)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.00001)  # Based on iteration % 3 == 2

# Training loop
num_epochs = 2
total_train_samples = 5000  # Limit training samples for speed
batch_count = 0
train_losses = []

model.train()
for epoch in range(num_epochs):
    running_loss = 0.0
    for inputs, labels in train_loader:
        if batch_count >= total_train_samples // batch_size:
            break
        
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        train_losses.append(loss.item())
        
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
        
        batch_count += 1
    print(f"Epoch {epoch+1}, Loss: {running_loss / (batch_count if batch_count > 0 else 1)}")

# Evaluation on validation data
all_preds = []
all_labels = []

model.eval()
with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Calculate AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Validation AUC: {final_auc}")

# Save metrics to JSON
metrics = {
    "final_train_loss": sum(train_losses) / len(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": total_train_samples,
    "eval_samples": batch_count * batch_size
}
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: The model uses 4 convolutional layers with batch normalization and ReLU activation. It employs adaptive average pooling to flatten the features, followed by two fully connected layers with dropout for regularization.

2. **Training Loop**: The loop runs for a maximum of 2 epochs, training on the first ~5000 samples (as specified). After each batch, it clears GPU cache to prevent memory fragmentation.

3. **Evaluation**: The model is evaluated on the same train loader in eval mode using AUC as the metric.

4. **Metrics and Checkpointing**: Training loss, validation AUC, number of parameters, epochs trained, batch size, training samples, and evaluation samples are saved to a JSON file and printed for logging purposes. The model state dictionary is also saved for future use or submission.
