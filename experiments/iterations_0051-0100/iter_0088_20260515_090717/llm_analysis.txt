```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Hyperparameters
epochs = 6
batch_size = 16  # Reduced for faster iteration
lr = 0.001      # Slightly increased learning rate
dropout_rate = 0.5
l2_reg = 0.0001  # Added L2 regularization

NUM_SPECIES = 206

# Define the CNN model
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),  # Added one more layer
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(128, 64), nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(64, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Model setup
model = BirdCLEFModel(NUM_SPECIES)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=l2_reg)

# Training loop
if torch.cuda.is_available():
    model.to('cuda')

total_train_loss = 0.0

for epoch in range(epochs):
    model.train()
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        total_train_loss += loss.item()
        print(f'Epoch {epoch+1}, Batch {batch_idx+1}, Loss: {loss.item()}')
        
        torch.cuda.empty_cache()  # Clear GPU cache

# Validation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Compute AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics and model checkpoint
metrics = {
    "final_train_loss": total_train_loss / (len(train_loader) * epochs),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": 15000,  # Assuming subset size
    "eval_samples": 5000       # Assuming subset size
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```
