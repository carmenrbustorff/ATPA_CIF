```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants
NUM_SPECIES = 206
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Hyperparameters based on iteration number (Example for Iteration 1)
iteration_number = 1  # This should be dynamically set in your script

# Batch size options
batch_size_options = [8, 16, 32]
batch_size = batch_size_options[iteration_number % 3]

# Learning rate options
lr_options = [0.001, 0.0001, 0.00001]
learning_rate = lr_options[iteration_number % 3]

# Architecture depth options (Number of Conv layers)
conv_depth_options = [2, 3, 4]
num_conv_layers = conv_depth_options[iteration_number % 3]

# Regularization options
dropout_rate = 0.2 if iteration_number == 1 else 0.5 if iteration_number == 2 else 0.0

# Load DataLoader
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Limit the dataset to 5000 samples for fast feedback
subset = torch.utils.data.Subset(train_loader.dataset, range(min(5000, len(train_loader.dataset))))
train_loader = torch.utils.data.DataLoader(subset, batch_size=batch_size, shuffle=True)

class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, dropout_rate=dropout_rate, num_conv_layers=num_conv_layers):
        super(BirdCLEFModel, self).__init__()
        
        layers = []
        in_channels = 1
        out_channels = 32
        
        # Add Conv-BN-ReLU layers
        for _ in range(num_conv_layers):
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
            in_channels = out_channels
            out_channels *= 2
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Classifier
        classifier_layers = [nn.Linear(in_channels, 128), nn.ReLU()]
        if dropout_rate > 0:
            classifier_layers.append(nn.Dropout(dropout_rate))
        classifier_layers.append(nn.Linear(128, num_species))
        self.classifier = nn.Sequential(*classifier_layers)
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, and optimizer
model = BirdCLEFModel(num_species=NUM_SPECIES, dropout_rate=dropout_rate, num_conv_layers=num_conv_layers).cuda()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Training loop
epochs = 6
total_train_loss = 0.0

for epoch in range(epochs):
    model.train()
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.cuda(), labels.cuda()
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        total_train_loss += loss.item()
        
        # Clear GPU cache
        torch.cuda.empty_cache()
        
        if (batch_idx + 1) % 50 == 0:
            print(f'Epoch [{epoch+1}/{epochs}], Step [{batch_idx+1}/{len(train_loader)}], Loss: {loss.item():.4f}')

final_train_loss = total_train_loss / len(train_loader)

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.cuda(), labels.cuda()
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

import numpy as np
from sklearn.metrics import roc_auc_score

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Metrics and model saving
metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(train_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:
1. **Imports**: The script starts with the required imports.
2. **Hyperparameters**: Hyperparameters are set based on the iteration number following the given rules for systematic exploration.
3. **DataLoader**: The DataLoader is loaded and limited to 5000 samples using `torch.utils.data.Subset`.
4. **Model Architecture**: A CNN model is defined with a flexible depth, batch normalization after each convolutional layer, and dropout in the fully connected layers.
5. **Training Loop**: The training loop uses Adam optimizer and BCELoss, with a focus on small-scale iterations for rapid experimentation.
6. **Evaluation**: The model is evaluated using the same DataLoader in eval mode, and AUC is computed using `roc_auc_score`.
7. **Metrics & Model Saving**: Metrics are saved to JSON and printed for logging, and the model's state is saved for further use or submission.

This setup prioritizes rapid experimentation by keeping training iterations small while maintaining a robust evaluation and metrics capture mechanism.
