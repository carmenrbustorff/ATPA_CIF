```python
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import models
from sklearn.metrics import roc_auc_score
import json
import numpy as np

# Constants
NUM_SPECIES = 206
BATCH_SIZE = 128
LR = 0.0001
DROPOUT_RATE = 0.2
MAX_EPOCHS = 20
EARLY_STOPPING_PATIENCE = 5

# Model definition
class BirdClassifier(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, dropout_rate=DROPOUT_RATE):
        super(BirdClassifier, self).__init__()
        self.base_model = models.efficientnet_b0(pretrained=True)
        self.base_model.classifier[1] = nn.Linear(self.base_model.classifier[1].in_features, 512)
        self.classifier = nn.Sequential(
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(512, num_species),
            nn.Sigmoid()
        )

    def forward(self, x):
        x = self.base_model(x)
        x = self.classifier(x)
        return x

# Load data
train_loader, val_loader = build_train_val_dataloaders(batch_size=BATCH_SIZE, augment=True)

# Initialize model, optimizer, and loss function
model = BirdClassifier()
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)
optimizer = optim.Adam(model.parameters(), lr=LR, weight_decay=0.001)
criterion = nn.BCELoss()

# Training loop
best_val_auc = 0
early_stopping_counter = 0

for epoch in range(MAX_EPOCHS):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item() * inputs.size(0)
    
    epoch_train_loss = running_loss / len(train_loader.dataset)
    print(f'Epoch {epoch+1}/{MAX_EPOCHS}, Train Loss: {epoch_train_loss:.4f}')
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            preds = (outputs > 0.5).float().cpu().numpy()
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())
    
    all_preds = np.array(all_preds)
    all_labels_one_hot = np.eye(NUM_SPECIES)[all_labels]
    val_auc = roc_auc_score(all_labels_one_hot, all_preds, average='weighted', multi_class='ovr')
    print(f'Validation AUC: {val_auc:.4f}')
    
    # Early stopping
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        early_stopping_counter += 1
        if early_stopping_counter >= EARLY_STOPPING_PATIENCE:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": epoch_train_loss,
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + str(metrics))
```
