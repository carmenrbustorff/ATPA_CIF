DROPOUT_OPTIONS = [0.0, 0.2, 0.5]
L2_REGULARIZATION_OPTIONS = [0, 0.0001, 0.001]

iteration_number = 1  # Change this to your current iteration number

# Select hyperparameters based on iteration number
batch_size = BATCH_SIZE_OPTIONS[iteration_number % 3]
lr = LEARNING_RATE_OPTIONS[iteration_number % 3]
dropout_rate = DROPOUT_OPTIONS[iteration_number % len(DROPOUT_OPTIONS)]
l2_regularization = L2_REGULARIZATION_OPTIONS[iteration_number % len(L2_REGULARIZATION_OPTIONS)]

# Define the model
class BirdClassifier(nn.Module):
    def __init__(self, num_species, dropout=0.5):
        super(BirdClassifier, self).__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1)
        self.bn1 = nn.BatchNorm2d(32)
        self.relu = nn.ReLU()
        self.pool = nn.MaxPool2d(kernel_size=2, stride=2, padding=0)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1)
        self.bn2 = nn.BatchNorm2d(64)
        self.adaptive_pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc = nn.Linear(64, num_species)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.relu(x)
        x = self.pool(x)
        
        x = self.conv2(x)
        x = self.bn2(x)
        x = self.relu(x)
        x = self.pool(x)
        
        x = self.adaptive_pool(x)
        x = x.view(x.size(0), -1)  # Flatten
        x = self.dropout(x)
        x = self.fc(x)
        return x

# Initialize model, optimizer, and criterion
model = BirdClassifier(NUM_SPECIES, dropout=dropout_rate).cuda()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=l2_regularization)
criterion = nn.BCEWithLogitsLoss()

# Load data
