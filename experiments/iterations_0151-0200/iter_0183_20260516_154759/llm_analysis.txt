##### D. Implement Early Stopping
Implement early stopping to prevent overfitting and save the best model during training.

```python
from sklearn.metrics import roc_auc_score

best_auc = 0
early_stopping_patience = 5
patience_counter = 0

for epoch in range(max_epochs):
    # Training loop
    model.train()
    total_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        with torch.amp.autocast("cuda"):
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
        
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        
        total_loss += loss.item()
    
    avg_train_loss = total_loss / len(train_loader)
    print(f"Epoch [{epoch+1}/{max_epochs}], Avg Train Loss: {avg_train_loss:.4f}")
    
    # Validation loop
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            
            outputs = model(inputs)
            preds = torch.sigmoid(outputs).cpu().numpy()
            all_preds.append(preds)
            all_labels.extend(labels.cpu().numpy())
    
    all_preds = np.vstack(all_preds)
    y_true_one_hot = np.zeros((len(all_labels), NUM_SPECIES))
    for row_idx, col_idx in enumerate(all_labels):
        y_true_one_hot[row_idx, col_idx] = 1.0
    
    valid_classes = np.any(y_true_one_hot == 1, axis=0)
    filtered_true = y_true_one_hot[:, valid_classes]
    filtered_preds = all_preds[:, valid_classes]
    
    final_auc = roc_auc_score(filtered_true, filtered_preds, average='macro', multi_class='ovr')
    print(f"Final Validation AUC: {final_auc:.4f}")
    
    # Early stopping
    if final_auc > best_auc:
        best_auc = final_auc
        patience_counter = 0
        torch.save(model.state_dict(), 'model_best.pt')
    else:
        patience_counter += 1
        if patience_counter >= early_stopping_patience:
            print("Early stopping triggered.")
            break
    
    scheduler.step()
```
