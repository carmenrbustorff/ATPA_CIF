### Updated Code

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torchvision import models
from sklearn.metrics import roc_auc_score
import numpy as np
import json

# Constants
NUM_SPECIES = 206
BATCH_SIZE = 256
LEARNING_RATE = 0.00001
DROPOUT = 0.5
EPOCHS = 20

# Model setup
base_model = models.resnet18(pretrained=True)
base_model.conv1 = nn.Conv2d(1, 64, kernel_size=(7, 7), stride=(2, 2), padding=(3, 3), bias=False)
num_ftrs = base_model.fc.in_features
base_model.fc = nn.Sequential(
    nn.Dropout(DROPOUT),
    nn.Linear(num_ftrs, NUM_SPECIES)
)

# Move model to CUDA
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = base_model.to(device)

# Loss and optimizer
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=0.001)

# Data loading (assuming build_train_val_dataloaders is correctly implemented)
from data_loader import build_train_val_dataloaders  # Assuming this function exists

train_loader, val_loader = build_train_val_dataloaders(batch_size=BATCH_SIZE, augment=True)

# Training loop
model.train()
for epoch in range(EPOCHS):
    running_loss = 0.0
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.float().to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item() * inputs.size(0)
    
    epoch_loss = running_loss / len(train_loader.dataset)
    print(f'Epoch {epoch+1}/{EPOCHS}, Loss: {epoch_loss:.4f}')

# Validation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to(device), labels.float().to(device)
        outputs = model(inputs)
        
        preds = torch.sigmoid(outputs).cpu().numpy()
        all_preds.extend(preds)
        all_labels.extend(labels.cpu().numpy())

all_preds = np.array(all_preds)
all_labels = np.eye(NUM_SPECIES)[all_labels]

final_auc = roc_auc_score(all_labels, all_preds, average='weighted', multi_class='ovr')
print(f'Final Validation AUC: {final_auc:.4f}')

# Metrics capture
metrics = {
    "final_train_loss": epoch_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

print("METRICS: ", json.dumps(metrics))

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```
