```python
import os
import json
import torch
import torch.nn as nn
import torch.optim as optim
from tqdm import tqdm
import numpy as np
from sklearn.metrics import roc_auc_score
from data_loader import build_train_val_dataloaders
import timm

# 1. Setup
NUM_SPECIES = 206
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# 2. Data Loaders
train_loader, val_loader = build_train_val_dataloaders(
    batch_size=128,
    augment=True,
    val_split=0.2
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# 3. Model Architecture (Transfer Learning)
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes):
        super().__init__()
        self.base_model = timm.create_model("efficientnet_b0", pretrained=True, in_chans=1, num_classes=0)
        
        # Unfreeze the base model weights
        for param in self.base_model.parameters():
            param.requires_grad = True  # Changed from False to True
                
        self.classifier = nn.Sequential(
            nn.Linear(int(self.base_model.num_features), 256), # type: ignore
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(256, num_classes)
        )
        
    def forward(self, x):
        features = self.base_model(x)
        logits = self.classifier(features)
        return logits

model = BirdCLEFModel(num_classes=NUM_SPECIES).to(device)

# 4. Optimizer, Loss, and AMP Scaler
criterion = nn.BCEWithLogitsLoss()

# Pass ALL parameters and lower the learning rate for fine-tuning
optimizer = optim.Adam(model.parameters(), lr=0.0001) 
scaler = torch.amp.GradScaler("cuda") # type: ignore
max_epochs = 3

# Training Loop
for epoch in range(1, max_epochs + 1):
    model.train()
    total_loss = 0.0
    
    for batch_idx, (inputs, targets) in enumerate(train_loader):
        inputs, targets = inputs.to(device), targets.to(device)
        
        optimizer.zero_grad()
        
        with torch.amp.autocast("cuda"):
            outputs = model(inputs)
            loss = criterion(outputs, targets.float())
        
        scaler.scale(loss).backward()
        
        # Unscale and clip gradients
        scaler.unscale_(optimizer)
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        
        scaler.step(optimizer)
        scaler.update()
        
        total_loss += loss.item()
    
    avg_train_loss = total_loss / len(train_loader)
    print(f"Epoch [{epoch}/{max_epochs}], Loss: {avg_train_loss:.4f}")
    
    # Validation Loop
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, targets in val_loader:
            inputs, targets = inputs.to(device), targets.to(device)
            
            with torch.amp.autocast("cuda"):
                outputs = model(inputs)
            
            preds = torch.sigmoid(outputs).cpu().numpy()
            labels = targets.cpu().numpy()
            
            all_preds.extend(preds)
            all_labels.extend(labels)
    
    all_preds = np.array(all_preds)
    all_labels = np.array(all_labels)

    # Drop columns that have no true positives in this validation slice
    valid_classes = np.any(all_labels == 1, axis=0)
    filtered_true = all_labels[:, valid_classes]
    filtered_preds = all_preds[:, valid_classes]

    if filtered_true.shape[1] > 0:
        final_auc = float(roc_auc_score(filtered_true, filtered_preds, average='macro', multi_class='ovr'))
        print(f"Final Validation AUC: {final_auc:.4f}")
    else:
        print("No valid classes in the current validation batch.")
    
# Save metrics.json and model.pt at the end of the script
metrics = {
    "train_loss": avg_train_loss,
    "validation_auc": final_auc if 'final_auc' in locals() else None
}

with open("metrics.json", "w") as f:
    json.dump(metrics, f)

torch.save(model.state_dict(), "model.pt")
```

   - Ensured that validation predictions and labels are correctly accumulated and filtered.
