```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import soundfile as sf

# Define hyperparameters
NUM_SPECIES = 206
BATCH_SIZE = 32  # Set based on iteration number, e.g., Iteration % 3 == 1: batch_size = 128
LR = 0.0001      # Learning rate based on iteration number
DROPOUT_RATE = 0.5  # Dropout rate for regularization
WEIGHT_DECAY = 1e-4  # L2 regularization weight decay

# Data ingestion and augmentation
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,      # Enable augmentation for training
    val_split=0.2,     # 80/20 split
)

# Define the model using transfer learning with EfficientNet
import timm
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes):
        super(BirdCLEFModel, self).__init__()
        self.base_model = timm.create_model('efficientnet_b0', pretrained=True)
        # Modify the first conv layer to accept 1-channel input (mono mel-spectrogram)
        self.base_model.conv_stem.in_channels = 1
        torch.nn.init.kaiming_uniform_(self.base_model.conv_stem.weight[:, :1])
        
        # Freeze the base model
        for param in self.base_model.parameters():
            param.requires_grad = False
        
        # Add a classifier head
        self.classifier = nn.Sequential(
            nn.AdaptiveAvgPool2d(1),
            nn.Flatten(),
            nn.Linear(self.base_model.num_features, 512),
            nn.ReLU(),
            nn.Dropout(DROPOUT_RATE),
            nn.Linear(512, num_classes)
        )
    
    def forward(self, x):
        features = self.base_model(x)
        logits = self.classifier(features)
        return logits

# Initialize the model, loss function, optimizer, and scheduler
model = BirdCLEFModel(NUM_SPECIES).to('cuda')
criterion = nn.BCEWithLogitsLoss()  # Use BCEWithLogitsLoss for raw logits
optimizer = optim.Adam(model.parameters(), lr=LR, weight_decay=WEIGHT_DECAY)
scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=10, gamma=0.1)

# Training loop
max_epochs = 20
early_stopping_patience = 5
best_val_auc = 0.0
no_improvement_count = 0

for epoch in range(max_epochs):
    model.train()
    total_loss = 0.0
    
    for inputs, labels in tqdm(train_loader):
        inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item()
    
    avg_train_loss = total_loss / len(train_loader)
    print(f'Epoch [{epoch+1}/{max_epochs}], Avg Train Loss: {avg_train_loss:.4f}')
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in tqdm(val_loader):
            inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
            outputs = model(inputs)
            preds = torch.sigmoid(outputs).cpu().numpy()
            
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())
    
    # Calculate AUC
    from sklearn.metrics import roc_auc_score
    all_preds = np.array(all_preds)
    all_labels = np.eye(NUM_SPECIES)[all_labels]
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    
    print(f'Validation AUC: {val_auc:.4f}')
    
    # Early stopping and model checkpointing
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        no_improvement_count = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        no_improvement_count += 1
    
    if no_improvement_count >= early_stopping_patience:
        print('Early stopping triggered.')
        break

# Save metrics to JSON
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: ", json.dumps(metrics))
```

This script defines a new model architecture using EfficientNet for transfer learning, with modifications to accept 1-channel input (mono mel-spectrogram). The training loop includes automatic mixed precision, batch normalization, and early stopping. Validation metrics are computed and saved to 'metrics.json', along with the trained model weights.
