import os

NUM_SPECIES = 206
BATCH_SIZE = 128
MAX_EPOCHS = 40
LEARNING_RATE = 0.0001
DROPOUT_RATE = 0.5
WEIGHT_DECAY = 1e-4

def build_model(num_classes):
    model = nn.Sequential(
        nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1),
        nn.ReLU(),
        nn.MaxPool2d(kernel_size=2, stride=2),
        nn.BatchNorm2d(32),

        nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1),
        nn.ReLU(),
        nn.MaxPool2d(kernel_size=2, stride=2),
        nn.BatchNorm2d(64),

        nn.AdaptiveAvgPool2d((1, 1)),
        nn.Flatten(),

        nn.Linear(64, 256),
        nn.ReLU(),
        nn.Dropout(DROPOUT_RATE),

        nn.Linear(256, num_classes)
    )
    return model

def train_model(model, train_loader, val_loader, criterion, optimizer, scheduler, max_epochs):
    scaler = torch.amp.GradScaler("cuda")
    early_stopping_patience = 5
    best_val_auc = 0
    epochs_no_improve = 0

    for epoch in range(max_epochs):
        model.train()
        running_loss = 0.0
        total_train_samples = 0

        for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{max_epochs}"):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')

            # SpecAugment
            freq_mask = nn.functional.dropout(torch.ones(inputs.shape[0], 1, inputs.shape[2], inputs.shape[3]), p=0.1)
            time_mask = nn.functional.dropout(torch.ones(inputs.shape[0], 1, inputs.shape[2], inputs.shape[3]), p=0.1)
            inputs = inputs * freq_mask * time_mask

            optimizer.zero_grad()

            with torch.cuda.amp.autocast():
                outputs = model(inputs)
                loss = criterion(outputs, labels)

            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()

            running_loss += loss.item() * inputs.size(0)
            total_train_samples += inputs.size(0)

        epoch_loss = running_loss / total_train_samples
        print(f"Epoch {epoch+1}/{max_epochs}, Train Loss: {epoch_loss:.4f}")

        model.eval()
        all_preds, all_labels = [], []

        with torch.no_grad():
            for inputs, labels in val_loader:
                inputs, labels = inputs.to('cuda'), labels.to('cuda')
                outputs = model(inputs)
                preds = torch.sigmoid(outputs).cpu().numpy()
                labels = labels.cpu().numpy()

        if val_auc > best_val_auc:
            best_val_auc = val_auc
            epochs_no_improve = 0
            torch.save(model.state_dict(), 'model.pt')
        else:
            epochs_no_improve += 1

        if epochs_no_improve == early_stopping_patience:
