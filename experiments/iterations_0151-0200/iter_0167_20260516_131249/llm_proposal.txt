```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import soundfile as sf

# Define hyperparameters
NUM_SPECIES = 206
BATCH_SIZE = 128  # Iteration % 3 == 1: lr = 0.0001 (reduced for fine-tuning)
LEARNING_RATE = 0.0001
MAX_EPOCHS = 40
DROPOUT_RATE = 0.5
WEIGHT_DECAY = 1e-4

# Data directories
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Build data loaders
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
    num_workers=4,
    pin_memory=True
)

# Load pre-trained model (EfficientNet B0 adapted for 1-channel)
import timm
model = timm.create_model('efficientnet_b0', pretrained=True, in_chans=1, num_classes=NUM_SPECIES)
for param in model.parameters():
    param.requires_grad = False

# Add a small classification head
model.classifier = nn.Sequential(
    nn.Dropout(DROPOUT_RATE),
    nn.Linear(model.classifier.in_features, NUM_SPECIES)
)

# Move model to CUDA
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Loss function and optimizer
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.AdamW(model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY)
scaler = torch.amp.GradScaler("cuda")

# Learning rate scheduler
scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, 'max', patience=3, factor=0.5, min_lr=1e-6)

# SpecAugment (for audio data)
from torchaudio.transforms import TimeStretch, PitchShift

stretch_transform = TimeStretch(rate=[0.9, 1.1])
pitch_shift_transform = PitchShift(sample_rate=44100, n_steps=[-2, 2])

# Training loop
best_auc = 0
early_stopping_patience = 5
epochs_since_improvement = 0

for epoch in range(MAX_EPOCHS):
    model.train()
    total_loss = 0
    
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{MAX_EPOCHS}"):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Apply SpecAugment
        if train_loader.dataset.augment:
            inputs = stretch_transform(inputs.squeeze(1)).unsqueeze(1)
            inputs = pitch_shift_transform(inputs.squeeze(1)).unsqueeze(1)
        
        optimizer.zero_grad()
        with torch.cuda.amp.autocast():
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
        
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        
        total_loss += loss.item()
    
    avg_train_loss = total_loss / len(train_loader)
    print(f"Epoch {epoch+1}/{MAX_EPOCHS}, Train Loss: {avg_train_loss:.4f}")
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in tqdm(val_loader, desc="Validation"):
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            preds = torch.sigmoid(outputs).cpu().numpy()
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())
    
    from sklearn.metrics import roc_auc_score
    all_preds = np.array(all_preds)
    all_labels = np.eye(NUM_SPECIES)[all_labels]
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted', multi_class='ovr')
    print(f"Validation AUC: {val_auc:.4f}")
    
    # Early stopping and learning rate scheduler
    if val_auc > best_auc:
        best_auc = val_auc
        epochs_since_improvement = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        epochs_since_improvement += 1
    
    if epochs_since_improvement >= early_stopping_patience:
        print("Early stopping triggered.")
        break
    
    scheduler.step(val_auc)

# Save metrics
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": best_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

print("METRICS: " + json.dumps(metrics))
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

```

This script defines a new model architecture using EfficientNet B0 adapted for 1-channel input (audio data). It includes SpecAugment transformations to augment the training data. The training loop uses Automatic Mixed Precision (AMP) for faster training and reduced memory usage. The validation AUC is calculated using `roc_auc_score` with `multi_class='ovr'`. The model's performance is monitored using early stopping, and metrics are saved to a JSON file.
