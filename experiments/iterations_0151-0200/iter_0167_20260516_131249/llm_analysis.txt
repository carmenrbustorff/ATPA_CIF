import torch
import torch.nn as nn
import torch.optim as optim
from timm import create_model
from sklearn.metrics import roc_auc_score
import numpy as np
import json

# Define constants
NUM_SPECIES = 206
BATCH_SIZE = 128
LEARNING_RATE = 0.0001
EPOCHS = 20

# Load data using build_train_val_dataloaders (assuming it's correctly implemented)
train_loader, val_loader = build_train_val_dataloaders(batch_size=BATCH_SIZE, augment=True)

# Initialize model
model = create_model('efficientnet_b0', pretrained=False, num_classes=NUM_SPECIES).to('cuda')

# Define loss function and optimizer
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE)

# Training loop
for epoch in range(EPOCHS):
    model.train()
    total_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
        
        # Apply SpecAugment (example implementation)
        # Assuming SpecAugment is applied directly to the input tensor
        # spec_augmented_inputs = apply_specaugment(inputs)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item()
    
    avg_train_loss = total_loss / len(train_loader)
    print(f'Epoch {epoch+1}/{EPOCHS}, Train Loss: {avg_train_loss:.4f}')

# Validation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
        outputs = model(inputs)
        
        # One-hot encode labels
        one_hot_labels = np.eye(NUM_SPECIES)[labels.cpu().numpy()]
        all_preds.extend(torch.sigmoid(outputs).cpu().numpy())
        all_labels.extend(one_hot_labels)

# Calculate validation AUC
all_preds = np.array(all_preds)
all_labels = np.array(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted', multi_class='ovr')
print(f'Final Validation AUC: {final_auc:.4f}')

# Save metrics to JSON
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: ", json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
