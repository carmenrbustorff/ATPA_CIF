import os

# Hyperparameters
max_epochs = 40
lr = 0.0001
batch_size = 32
dropout_rate = 0.2

class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(128, 256), nn.ReLU(), nn.Dropout(dropout_rate),
            nn.Linear(256, num_species)  # NO BatchNorm1d!
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return self.classifier(x)

# Data loading
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=batch_size,
    augment=True,      # Enable augmentation for training
    val_split=0.2,     # 80/20 split
)

# Model setup
model = BirdCLEFModel(NUM_SPECIES).to('cuda')
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.Adam(model.parameters(), lr=lr)
scaler = torch.cuda.amp.GradScaler()

# Training loop
train_losses = []
early_stopping_counter = 0
best_auc = 0.0

for epoch in range(max_epochs):
    model.train()
    epoch_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
        
        with torch.cuda.amp.autocast("cuda"):
            outputs = model(inputs)
            loss = criterion(outputs, labels)
        
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        
        epoch_loss += loss.item()
        train_losses.append(loss.item())
        
        if (batch_idx + 1) % 10 == 0:
            print(f"Epoch [{epoch+1}/{max_epochs}], Step [{batch_idx+1}/{len(train_loader)}], Loss: {loss.item():.4f}")
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.float().to('cuda')
            outputs = model(inputs)
            preds = torch.sigmoid(outputs).cpu().numpy()
            all_preds.extend(preds)

### Explanation

1. **Model Architecture**: 
   - A simple CNN with 2 convolutional layers followed by a fully connected layer.
   - Uses `BCEWithLogitsLoss` which combines a sigmoid function and BCE loss, avoiding the need to apply sigmoid manually in the forward pass.

2. **Data Loading**:
   - Uses the provided data loader to get training and validation datasets.
   - Sets up the model, optimizer, and scaler for automatic mixed precision (AMP).

3. **Training Loop**:
   - Trains the model for a specified number of epochs.
   - Uses AMP to optimize memory usage and speed.
   - Computes loss per batch and prints it.

4. **Validation**:
   - Evaluates the model on the validation set after each epoch.
   - Computes AUC using `roc_auc_score` with `multi_class='ovr'`.

5. **Early Stopping**:
   - Stops training if the validation AUC does not improve for 5 consecutive epochs.

6. **Metrics and Model Saving**:
   - Saves final metrics to a JSON file.
   - Saves the best model state dictionary to `model.pt`.

This setup is designed for rapid experimentation with different hyperparameters and architectures, allowing you to quickly iterate and find improvements in your model's performance.
