```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import soundfile as sf

# Define the CNN model architecture
class BirdClassifier(nn.Module):
    def __init__(self, num_classes, depth=3, dropout_rate=0.2):
        super(BirdClassifier, self).__init__()
        channels = [32, 64, 128, 256][:depth]
        layers = []
        
        for i in range(depth):
            if i == 0:
                layers.append(nn.Conv2d(1, channels[i], kernel_size=3, padding=1))
            else:
                layers.append(nn.Conv2d(channels[i-1], channels[i], kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(channels[i]))
            layers.append(nn.ReLU())
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_classes),
            nn.Sigmoid()
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return self.classifier(x)

# Hyperparameters
max_epochs = 40
batch_size_options = {0: 8, 1: 16, 2: 32}
lr_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
depth_options = {0: 2, 1: 3, 2: 4}
dropout_options = [0.0, 0.2, 0.5]

# Define the training loop
def train_model(iteration):
    # Determine hyperparameters based on iteration number
    iteration_mod = iteration % 3
    batch_size = batch_size_options[iteration_mod]
    lr = lr_options[iteration_mod]
    depth = depth_options[iteration_mod]
    dropout_rate = dropout_options[iteration_mod]

    # Data loading
    DATA_DIR = "/mnt/disks/data/birdclef"
    METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
    train_loader, val_loader = build_train_val_dataloaders(
        metadata_csv=METADATA_FILE,
        audio_dir=DATA_DIR,
        batch_size=batch_size,
        augment=True,
        val_split=0.2,
        num_workers=4,
        pin_memory=True
    )
    
    # Model setup
    model = BirdClassifier(num_classes=train_loader.dataset.dataset.num_classes, depth=depth, dropout_rate=dropout_rate)
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model.to(device)
    
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)
    scaler = torch.cuda.amp.GradScaler()
    scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=10, gamma=0.1)
    
    # Training loop
    for epoch in range(max_epochs):
        model.train()
        total_loss = 0
        
        for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{max_epochs}"):
            inputs, labels = inputs.to(device), labels.to(device)
            
            optimizer.zero_grad()
            with torch.cuda.amp.autocast():
                outputs = model(inputs)
                loss = criterion(outputs, labels.float())
            
            scaler.scale(loss).backward()
            scaler.step(optimizer)
            scaler.update()
            
            total_loss += loss.item()
        
        print(f"Epoch {epoch+1}/{max_epochs}, Loss: {total_loss / len(train_loader)}")
        scheduler.step()
    
    # Evaluation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in tqdm(val_loader):
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    
    # Compute AUC
    from sklearn.metrics import roc_auc_score
    final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Final Validation AUC: {final_auc}")
    
    # Save metrics and model
    metrics = {
        "final_train_loss": total_loss / len(train_loader),
        "final_auc": final_auc,
        "num_params": sum(p.numel() for p in model.parameters()),
        "epochs_trained": max_epochs,
        "batch_size": batch_size,
        "training_samples": len(train_loader.dataset),
        "eval_samples": len(val_loader.dataset)
    }
    
    with open('metrics.json', 'w') as f:
        json.dump(metrics, f)
    
    print("METRICS: " + json.dumps(metrics))
    
    torch.save(model.state_dict(), 'model.pt')

# Run the training loop for a specific iteration
train_model(iteration=154)
```

This script defines a new CNN model architecture with varying hyperparameters based on the iteration number. It includes a training loop that uses mixed precision for faster convergence and evaluates the model using AUC. The metrics are saved to a JSON file, and the model weights are saved for later use.
