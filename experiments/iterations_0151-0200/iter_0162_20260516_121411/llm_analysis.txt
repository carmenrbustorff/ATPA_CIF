```python
import os
import json
import numpy as np
from sklearn.metrics import roc_auc_score
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, RandomSampler
from data_loader_module import build_train_val_dataloaders  # Replace with actual module

# Constants
NUM_SPECIES = 206
MAX_EPOCHS = 20
EARLY_STOPPING_PATIENCE = 5
CHECKPOINT_PATH = 'model.pt'
METRICS_PATH = 'metrics.json'

# Define the CNN model
class BirdCLEFModel(nn.Module):
    def __init__(self, num_layers=3, dropout_rate=0.2):
        super(BirdCLEFModel, self).__init__()
        layers = [
            nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)
        ]
        
        for _ in range(num_layers - 1):
            layers.extend([
                nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1),
                nn.BatchNorm2d(64),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
        
        layers.append(nn.AdaptiveAvgPool2d((1, 1)))
        layers.append(nn.Flatten())
        layers.append(nn.Linear(64 * 1 * 1, NUM_SPECIES))
        
        self.model = nn.Sequential(*layers)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        x = self.model(x)
        x = self.dropout(x)
        return x

# Training function
def train_model(iteration):
    # Hyperparameter selection based on iteration number
    if iteration % 3 == 0:
        batch_size = 8
        lr = 0.001
        num_layers = 2
        dropout_rate = 0.0
    elif iteration % 3 == 1:
        batch_size = 16
        lr = 0.0001
        num_layers = 3
        dropout_rate = 0.2
    else:
        batch_size = 32
        lr = 0.00001
        num_layers = 4
        dropout_rate = 0.5

    # Load dataset
    train_loader, val_loader = build_train_val_dataloaders(batch_size=batch_size, augment=True)

    # Initialize model, loss function, and optimizer
    model = BirdCLEFModel(num_layers=num_layers, dropout_rate=dropout_rate).to('cuda')
    criterion = nn.BCEWithLogitsLoss()
    optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.001)

    # Training loop
    best_auc = 0.0
    early_stopping_counter = 0

    for epoch in range(MAX_EPOCHS):
        model.train()
        total_loss = 0.0

        for inputs, labels in train_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            
            # Forward pass
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            
            # Backward pass and optimization
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()

        train_loss = total_loss / len(train_loader)
        print(f'Epoch {epoch+1}/{MAX_EPOCHS}, Train Loss: {train_loss:.4f}')

        # Validation
        model.eval()
        all_preds = []
        all_labels = []

        with torch.no_grad():
            for inputs, labels in val_loader:
                inputs, labels = inputs.to('cuda'), labels.to('cuda')
                outputs = model(inputs)
                
                preds = torch.sigmoid(outputs).cpu().numpy()
                all_preds.extend(preds)
                all_labels.extend(labels.cpu().numpy())

        # One-hot encode labels
        one_hot_labels = np.eye(NUM_SPECIES)[all_labels]
        
        val_auc = roc_auc_score(one_hot_labels, all_preds, multi_class='ovr')
        print(f'Validation AUC: {val_auc:.4f}')

        if val_auc > best_auc:
            best_auc = val_auc
            early_stopping_counter = 0
            torch.save(model.state_dict(), CHECKPOINT_PATH)
        else:
            early_stopping_counter += 1

        if early_stopping_counter >= EARLY_STOPPING_PATIENCE:
            print("Early stopping triggered.")
            break

    # Save metrics
    num_params = sum(p.numel() for p in model.parameters())
    metrics = {
        "final_train_loss": train_loss,
        "final_auc": best_auc,
        "num_params": num_params,
        "epochs_trained": epoch + 1,
        "batch_size": batch_size,
        "training_samples": len(train_loader.dataset),
        "eval_samples": len(val_loader.dataset)
    }
    
    print("METRICS: " + json.dumps(metrics))
    with open(METRICS_PATH, 'w') as f:
        json.dump(metrics, f)

# Example usage
iteration = 2  # Replace with actual iteration number
train_model(iteration)
```
