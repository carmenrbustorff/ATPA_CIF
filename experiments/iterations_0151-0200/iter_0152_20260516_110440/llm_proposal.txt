```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import torchaudio.transforms as T

# Hyperparameters
NUM_EPOCHS = 20
LEARNING_RATE = 0.001
BATCH_SIZE = 32
DROPOUT_RATE = 0.5
L2_REGULARIZATION = 1e-4
NUM_SPECIES = 206

class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES):
        super(BirdCLEFModel, self).__init__()
        
        # Define convolutional layers with increasing channels
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU()
        )
        
        # Adaptive pooling and flattening
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Classifier with dropout for regularization
        self.classifier = nn.Sequential(
            nn.Linear(128, 128), nn.Dropout(DROPOUT_RATE), nn.ReLU(),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
    num_workers=4,
    pin_memory=True
)

# Model setup
model = BirdCLEFModel(num_species=NUM_SPECIES).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=L2_REGULARIZATION)

# Training loop
best_val_auc = 0.0
patience = 5
early_stopping_counter = 0

for epoch in range(NUM_EPOCHS):
    model.train()
    running_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        torch.cuda.empty_cache()  # Clear GPU memory
        
    print(f"Epoch [{epoch+1}/{NUM_EPOCHS}], Loss: {running_loss/len(train_loader):.4f}")
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            preds = (outputs > 0.5).float().cpu().numpy()
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())
    
    from sklearn.metrics import roc_auc_score
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Validation AUC: {val_auc:.4f}")
    
    # Early stopping and model saving
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        early_stopping_counter += 1
    
    if early_stopping_counter >= patience:
        print("Early stopping triggered.")
        break

# Metrics saving
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Explanation:

1. **Model Architecture**:
   - The model consists of 3 convolutional layers followed by batch normalization and ReLU activation.
   - An adaptive average pooling layer is used to flatten the feature maps.
   - A classifier with dropout for regularization is used to output probabilities.

2. **Data Loading**:
   - The `build_train_val_dataloaders` function is used to load data with appropriate augmentations, shuffling, and parallel loading.

3. **Training Loop**:
   - The model is trained for 20 epochs with Adam optimizer and L2 regularization.
   - Loss is computed using BCELoss and sigmoid activation in the forward pass ensures outputs are in [0,1].

4. **Validation and Early Stopping**:
   - Validation AUC is computed after each epoch using `roc_auc_score`.
   - Early stopping is implemented to halt training if validation AUC does not improve for 5 consecutive epochs.

5. **Metrics and Model Saving**:
   - Metrics including final training loss, best validation AUC, number of parameters, etc., are saved to a JSON file.
   - The model with the best validation performance is saved as `model.pt`.

This setup allows for rapid experimentation with different hyperparameters while maintaining a systematic approach to model development.
