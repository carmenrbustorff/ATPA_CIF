import torch.hub

def load_model():
    try:
        # Attempt to load the model with trust options
        model = torch.hub.load('NVIDIA/DeepLearningExamples:torch/vision/main/classification/efficientnet', 'efficientnet_b0', trust_repo=True)
        return model
    except Exception as e:
        print(f"Error loading model: {e}")
        raise

class BirdClassifier(torch.nn.Module):
    def __init__(self, num_classes):
        super(BirdClassifier, self).__init__()
        self.base_model = load_model()
        self.flatten = torch.nn.AdaptiveAvgPool2d(1)
        self.fc = torch.nn.Linear(self.base_model.classifier[1].in_features, num_classes)

    def forward(self, x):
        x = self.base_model.features(x)
        x = self.flatten(x).squeeze(-1).squeeze(-1)
        x = self.fc(x)
        return x
