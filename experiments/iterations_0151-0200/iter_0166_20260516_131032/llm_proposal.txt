```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import soundfile as sf
import numpy as np
from sklearn.metrics import roc_auc_score

# Hyperparameters
NUM_SPECIES = 206
BATCH_SIZE = 32
LR = 0.001
MAX_EPOCHS = 40
DROPOUT_RATE = 0.5
WEIGHT_DECAY = 1e-4

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

# Model architecture
class BirdClassifier(nn.Module):
    def __init__(self, num_species):
        super(BirdClassifier, self).__init__()
        # Using EfficientNet as a base model
        self.base_model = torch.hub.load('NVIDIA/DeepLearningExamples:torch/vision/main/classification/efficientnet', 'efficientnet_b0')
        # Modify the first conv layer to accept 1 channel (mono mel-spectrogram)
        self.base_model.conv_stem[0] = nn.Conv2d(1, 32, kernel_size=(3, 3), stride=(2, 2), padding=(1, 1), bias=False)
        # Freeze the base model
        for param in self.base_model.parameters():
            param.requires_grad = False
        
        # Classification head
        self.classifier_head = nn.Sequential(
            nn.AdaptiveAvgPool2d(1),
            nn.Flatten(),
            nn.Linear(self.base_model.fc.in_features, 512),
            nn.ReLU(),
            nn.Dropout(DROPOUT_RATE),
            nn.Linear(512, num_species)
        )
    
    def forward(self, x):
        x = self.base_model(x)
        x = self.classifier_head(x)
        return x

model = BirdClassifier(NUM_SPECIES).to('cuda')

# Loss and optimizer
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.Adam(model.parameters(), lr=LR, weight_decay=WEIGHT_DECAY)

# Training loop
def train_epoch(model, dataloader, criterion, optimizer):
    model.train()
    total_loss = 0.0
    for inputs, labels in tqdm(dataloader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        # Apply SpecAugment directly to the batch tensors
        # For simplicity, let's assume a basic time masking here
        if np.random.rand() < 0.5:
            mask_width = torch.randint(1, 16, (1,))
            for i in range(inputs.size(0)):
                mask_start = torch.randint(0, inputs.size(2) - mask_width, (1,))
                inputs[i, :, mask_start:mask_start + mask_width] = 0
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item()
    return total_loss / len(dataloader)

# Evaluation
def evaluate(model, dataloader):
    model.eval()
    all_preds = []
    all_labels = []
    with torch.no_grad():
        for inputs, labels in tqdm(dataloader):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            preds = torch.sigmoid(outputs).cpu().numpy()
            all_preds.append(preds)
            all_labels.append(labels.cpu().numpy())
    
    all_preds = np.vstack(all_preds)
    all_labels = np.vstack(all_labels)
    
    # One-hot encode labels for AUC calculation
    label_binarizer = sklearn.preprocessing.LabelBinarizer()
    label_binarizer.fit(range(NUM_SPECIES))
    one_hot_labels = label_binarizer.transform(np.argmax(all_labels, axis=1))
    
    auc = roc_auc_score(one_hot_labels, all_preds, average='weighted', multi_class='ovr')
    return auc

# Early stopping
best_auc = 0.0
patience = 5
epochs_without_improvement = 0

for epoch in range(MAX_EPOCHS):
    train_loss = train_epoch(model, train_loader, criterion, optimizer)
    val_auc = evaluate(model, val_loader)
    
    print(f"Epoch {epoch + 1}/{MAX_EPOCHS}, Train Loss: {train_loss:.4f}, Val AUC: {val_auc:.4f}")
    
    if val_auc > best_auc:
        best_auc = val_auc
        epochs_without_improvement = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        epochs_without_improvement += 1
    
    if epochs_without_improvement >= patience:
        print("Early stopping triggered.")
        break

# Save metrics
metrics = {
    "final_train_loss": train_loss,
    "final_auc": best_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

This script defines a new model architecture using EfficientNet for the base model, with modifications to accept 1-channel input. The training loop includes SpecAugment for data augmentation and uses BCEWithLogitsLoss to handle raw logits from the model. Early stopping is implemented based on validation AUC, and metrics are saved in JSON format along with the model weights.
