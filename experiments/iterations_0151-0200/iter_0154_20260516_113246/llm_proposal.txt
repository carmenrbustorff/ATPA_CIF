```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import soundfile as sf

# Define hyperparameters based on the iteration number
iteration = 2  # Example iteration number, modify as needed
batch_size_options = {0: 8, 1: 16, 2: 32}
lr_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
conv_layers_options = {0: 2, 1: 3, 2: 4}
dropout_options = [0.0, 0.2, 0.5]
batch_norm_placement_options = ['after_conv', 'everywhere', 'sparse']

# Select options based on iteration number
batch_size = batch_size_options[iteration % 3]
lr = lr_options[iteration % 3]
num_conv_layers = conv_layers_options[iteration % 3]
dropout_rate = dropout_options[(iteration // 3) % len(dropout_options)]
batch_norm_placement = batch_norm_placement_options[(iteration // (3 * len(dropout_options))) % len(batch_norm_placement_options)]

# Define the CNN model
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes):
        super(BirdCLEFModel, self).__init__()
        conv_layers = []
        for i in range(num_conv_layers):
            in_channels = 1 if i == 0 else 32 * (2 ** (i - 1))
            out_channels = 32 * (2 ** i)
            conv_layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            if batch_norm_placement == 'everywhere' or (batch_norm_placement == 'after_conv' and i < num_conv_layers - 1):
                conv_layers.append(nn.BatchNorm2d(out_channels))
            conv_layers.append(nn.ReLU())
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(32 * (2 ** (num_conv_layers - 1)), 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Load data
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=batch_size,
    augment=True,
    val_split=0.2,
    num_workers=4,
    pin_memory=True
)

# Initialize model, optimizer, and loss function
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
num_classes = train_loader.dataset.num_classes
model = BirdCLEFModel(num_classes).to(device)
optimizer = optim.Adam(model.parameters(), lr=lr)
criterion = nn.BCELoss()

# Training loop
max_epochs = 10
final_train_loss = None
all_preds = []
all_labels = []

for epoch in range(max_epochs):
    model.train()
    train_loss = 0.0
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{max_epochs}"):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        
        # Backward and optimize
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        train_loss += loss.item() * inputs.size(0)
    
    train_loss /= len(train_loader.dataset)
    print(f"Epoch {epoch+1}/{max_epochs}, Train Loss: {train_loss:.4f}")

# Validation loop
model.eval()
with torch.no_grad():
    for inputs, labels in tqdm(val_loader, desc="Validation"):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Forward pass
        outputs = model(inputs)
        
        all_preds.append(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Compute AUC
from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds)
all_labels_one_hot = np.eye(num_classes)[all_labels]
final_auc = roc_auc_score(all_labels_one_hot, all_preds, average='weighted')
print(f"Final Validation AUC: {final_auc:.4f}")

# Save metrics and model
metrics = {
    "final_train_loss": train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": max_epochs,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: ", json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```

This script defines a CNN model for the BirdCLEF dataset, varying hyperparameters based on the iteration number. It includes a training loop with automatic GPU usage, validation to compute AUC, and saving of metrics and model checkpoints. Adjust the `iteration` variable to test different configurations systematically.
