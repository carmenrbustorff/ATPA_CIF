```python
import os
import json
import torch
import torch.nn as nn
import torch.optim as optim
from tqdm import tqdm
import numpy as np
from sklearn.metrics import roc_auc_score
from data_loader import build_train_val_dataloaders
import timm

# 1. Setup
NUM_SPECIES = 206
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# 2. Data Loaders
batch_size = 128
val_split = 0.2

train_loader, val_loader = build_train_val_dataloaders(
    batch_size=batch_size,
    augment=True,
    val_split=val_split
)

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# 3. Model Architecture (Transfer Learning)
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes):
        super().__init__()
        self.base_model = timm.create_model("efficientnet_b0", pretrained=True, in_chans=1, num_classes=0)
        
        # Freeze the base model weights
        for param in self.base_model.parameters():
            param.requires_grad = False
        
        self.classifier = nn.Sequential(
            nn.Linear(self.base_model.num_features, 256),
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(256, num_classes)
        )
        
    def forward(self, x):
        features = self.base_model(x)
        logits = self.classifier(features)
        return logits

model = BirdCLEFModel(num_classes=NUM_SPECIES).to(device)

# 4. Optimizer, Loss, and AMP Scaler
criterion = nn.BCEWithLogitsLoss()

# Pass ALL parameters and lower the learning rate for fine-tuning
optimizer = optim.Adam(model.parameters(), lr=0.0001) 
scaler = torch.amp.GradScaler("cuda")
max_epochs = 3

# Training loop
for epoch in range(max_epochs):
    model.train()
    train_loss = []
    
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}/{max_epochs}"):
        inputs = inputs.to(device)
        labels = labels.float().to(device)  # Ensure labels are float for BCEWithLogitsLoss
        
        optimizer.zero_grad()
        
        with torch.amp.autocast("cuda"):
            outputs = model(inputs)
            loss = criterion(outputs, labels)
        
        scaler.scale(loss).backward()
        
        # Unscale the gradients before clipping
        scaler.unscale_(optimizer)
        torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)
        
        scaler.step(optimizer)
        scaler.update()
        
        train_loss.append(loss.item())
    
    avg_train_loss = np.mean(train_loss)
    print(f"Epoch {epoch+1}/{max_epochs}, Average Train Loss: {avg_train_loss:.4f}")

# Validation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(val_loader, desc="Validation"):
        inputs = inputs.to(device)
        labels = labels.float().to(device)  # Ensure labels are float for BCEWithLogitsLoss
        
        with torch.amp.autocast("cuda"):
            outputs = model(inputs)
        
        preds = torch.sigmoid(outputs).cpu().numpy()
        all_preds.extend(preds)
        all_labels.extend(labels.cpu().numpy())

# Convert labels to one-hot encoding
all_preds = np.array(all_preds)
y_true_one_hot = np.zeros((len(all_labels), NUM_SPECIES))
for row_idx, col_idx in enumerate(all_labels):
    y_true_one_hot[row_idx, int(col_idx)] = 1.0

# Calculate final validation AUC using macro-averaged multi_class='ovr'
final_auc = roc_auc_score(y_true_one_hot, all_preds, average='macro', multi_class='ovr')
print(f"Final Validation AUC: {final_auc:.4f}")

# Save metrics to JSON
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": max_epochs,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```
