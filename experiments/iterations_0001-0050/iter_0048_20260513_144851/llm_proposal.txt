```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Define the CNN model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.dropout = nn.Dropout(0.2)
        self.classifier = nn.Sequential(
            nn.Linear(128, 128), nn.ReLU(),
            nn.Linear(128, num_classes)  # No BatchNorm1d!
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        x = self.dropout(x)
        return torch.sigmoid(self.classifier(x))

# Hyperparameters for Iteration % 3 == 1
batch_size = 16
lr = 0.0001
num_epochs = 2
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Initialize model, loss function, and optimizer
model = BirdCLEFModel().to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.0001)

# Training loop
model.train()
batch_count = 0
total_loss = 0.0
for epoch in range(num_epochs):
    for inputs, labels in tqdm(train_loader):
        if batch_count >= 10:  # Limit to first ~5000 samples (~10 batches)
            break

        inputs, labels = inputs.to('cuda'), labels.to('cuda')

        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()

        total_loss += loss.item()
        print(f"Epoch {epoch+1}/{num_epochs}, Batch {batch_count+1}, Loss: {loss.item()}")
        
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
        batch_count += 1

# Validation loop
model.eval()
all_preds = []
all_labels = []
with torch.no_grad():
    for inputs, labels in tqdm(train_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Compute validation AUC
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Collect metrics
num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)
metrics = {
    "final_train_loss": total_loss / batch_count,
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": batch_count * batch_size,
    "eval_samples": len(all_labels)
}

# Save model and metrics
torch.save(model.state_dict(), 'model.pt')
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

# Print metrics for log parsing
print("METRICS: " + json.dumps(metrics))
```

- **Training Loop**: We train for 2 epochs with a batch size of 16 and a learning rate of 0.0001. The training loop includes a dropout layer set to 0.2 for regularization.
- **Validation Loop**: After training, the model is evaluated on the same dataloader in eval mode to compute the validation AUC score.
- **Metrics Collection**: We collect and save metrics including final training loss, validation AUC, number of parameters, epochs trained, batch size, and sample counts for both training and evaluation.
- **Model Saving**: The model weights are saved to `model.pt`, and the metrics are saved to `metrics.json`.
