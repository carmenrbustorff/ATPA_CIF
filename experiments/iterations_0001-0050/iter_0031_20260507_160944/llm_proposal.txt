It also includes a linear layer for classification with a sigmoid activation function suitable for multi-label classification.

```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=32)

class BirdClassifier(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdClassifier, self).__init__()
        self.conv1 = nn.Conv2d(1, 64, kernel_size=(3, 3), stride=1, padding=1)
        self.bn1 = nn.BatchNorm2d(64)
        self.pool = nn.MaxPool2d(kernel_size=(2, 2), stride=2, padding=0)
        
        self.conv2 = nn.Conv2d(64, 128, kernel_size=(3, 3), stride=1, padding=1)
        self.bn2 = nn.BatchNorm2d(128)
        
        self.conv3 = nn.Conv2d(128, 256, kernel_size=(3, 3), stride=1, padding=1)
        self.bn3 = nn.BatchNorm2d(256)
        
        self.adaptive_pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc = nn.Linear(256, num_classes)

    def forward(self, x):
        x = self.pool(F.relu(self.bn1(self.conv1(x))))
        x = self.pool(F.relu(self.bn2(self.conv2(x))))
        x = self.pool(F.relu(self.bn3(self.conv3(x))))
        x = self.adaptive_pool(x)
        x = x.view(x.size(0), -1)  # Flatten
        x = torch.sigmoid(self.fc(x))
        return x

# Initialize model, loss function, and optimizer
model = BirdClassifier().to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Training loop
num_epochs = 3
total_samples = 5000
steps_per_epoch = total_samples // train_loader.batch_size

metrics = {
    'final_train_loss': None,
    'final_auc': None,  # Placeholder for AUC calculation if implemented
    'num_params': sum(p.numel() for p in model.parameters()),
    'epochs_trained': num_epochs,
    'batch_size': train_loader.batch_size
}

model.train()
for epoch in range(num_epochs):
    running_loss = 0.0
    for i, (inputs, labels) in enumerate(train_loader):
        if i >= steps_per_epoch:
            break
        
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
    
    epoch_loss = running_loss / steps_per_epoch
    print(f"Epoch [{epoch+1}/{num_epochs}], Loss: {epoch_loss:.4f}")
    metrics['final_train_loss'] = epoch_loss
    
    torch.cuda.empty_cache()

# Save model
torch.save(model.state_dict(), 'model.pt')

# Print and save metrics
print("METRICS:", json.dumps(metrics))
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)
```

   - **Adaptive Average Pooling**: Used to flatten the feature maps into a fixed-size vector, making the model more robust to varying input sizes.
   - **Linear Layer**: A fully connected layer with sigmoid activation for multi-label classification.
