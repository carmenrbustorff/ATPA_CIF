import os

batch_size_options = {0: 8, 1: 16, 2: 32}

batch_size = batch_size_options[iteration % 3]

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=3, dropout_rate=0.5, batch_norm_placement='everywhere'):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for i in range(num_conv_layers):
            out_channels = 32 * (2 ** i)
            conv_layer = nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=1, padding=1)
            if batch_norm_placement == 'everywhere' or (batch_norm_placement == 'after_conv' and i > 0):
                layers.append(nn.BatchNorm2d(out_channels))
            layers.extend([conv_layer, nn.ReLU(), nn.MaxPool2d(kernel_size=2, stride=2)])
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        
        if batch_norm_placement == 'everywhere':
            layers.append(nn.BatchNorm2d(in_channels))

            nn.AdaptiveAvgPool2d((1, 1)),
            nn.Flatten(),
            nn.Dropout(dropout_rate),
            nn.Linear(in_channels, num_classes)
        ])
        self.classifier = nn.Sequential(*layers)
