Given the constraints of the BirdCLEF+ 2026 Track B competition, we need to prioritize an efficient model that can be quickly iterated upon. We'll use a simple yet effective CNN architecture suitable for multi-label classification tasks on mel-spectrograms. This model will be implemented using TensorFlow/Keras and is designed to run efficiently even with limited computational resources.

Here's the proposed model architecture along with data loading, training loop, and evaluation code:

```python
import tensorflow as tf
from tensorflow.keras.layers import Conv2D, BatchNormalization, Activation, GlobalAveragePooling2D, Dense, Dropout
from tensorflow.keras.models import Model
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.losses import BinaryCrossentropy
from tensorflow.keras.metrics import AUC
from tensorflow.data import Dataset

# Define the model architecture
def create_model(input_shape, num_classes):
    inputs = tf.keras.Input(shape=input_shape)

    # First convolutional block
    x = Conv2D(32, (3, 3), padding='same')(inputs)
    x = BatchNormalization()(x)
    x = Activation('relu')(x)

    # Second convolutional block
    x = Conv2D(64, (3, 3), padding='same')(x)
    x = BatchNormalization()(x)
    x = Activation('relu')(x)
    x = tf.keras.layers.MaxPooling2D((2, 2))(x)

    # Third convolutional block
    x = Conv2D(128, (3, 3), padding='same')(x)
    x = BatchNormalization()(x)
    x = Activation('relu')(x)

    # Global Average Pooling and Dropout
    x = GlobalAveragePooling2D()(x)
    x = Dropout(0.5)(x)

    # Output layer
    outputs = Dense(num_classes, activation='sigmoid')(x)

    model = Model(inputs, outputs)
    return model

# Data loading function (assumes data is preprocessed into TFRecord format)
def load_dataset(file_pattern, batch_size):
    def parse_tfrecord(example):
        feature_description = {
            'mel_spectrogram': tf.io.FixedLenFeature([128, 30], tf.float32),
            'labels': tf.io.FixedLenFeature([234], tf.int64)
        }
        example = tf.io.parse_single_example(example, feature_description)
        mel_spectrogram = example['mel_spectrogram']
        labels = tf.cast(example['labels'], tf.float32)
        return mel_spectrogram, labels

    dataset = Dataset.list_files(file_pattern, shuffle=True)
    dataset = dataset.interleave(lambda x: tf.data.TFRecordDataset(x), cycle_length=8, num_parallel_calls=tf.data.AUTOTUNE)
    dataset = dataset.map(parse_tfrecord, num_parallel_calls=tf.data.AUTOTUNE)
    dataset = dataset.batch(batch_size).prefetch(tf.data.AUTOTUNE)
    return dataset

# Training loop
def train_model(model, train_dataset, val_dataset, epochs=10):
    model.compile(optimizer=Adam(learning_rate=0.001),
                  loss=BinaryCrossentropy(),
                  metrics=[AUC(multi_label=True)])

    history = model.fit(train_dataset,
                        validation_data=val_dataset,
                        epochs=epochs)
    return history

# Evaluation function
def evaluate_model(model, test_dataset):
    results = model.evaluate(test_dataset)
    print(f"Test Loss: {results[0]}, Test AUC: {results[1]}")

# Example usage
if __name__ == "__main__":
    # Define paths and parameters
    train_file_pattern = '/home/carme/birdclef-data/train/*.tfrecord'
    val_file_pattern = '/home/carme/birdclef-data/val/*.tfrecord'
    test_file_pattern = '/home/carme/birdclef-data/test/*.tfrecord'
    input_shape = (128, 30, 1)  # Assuming mel-spectrogram shape
    num_classes = 234
    batch_size = 64

    # Create the model
    model = create_model(input_shape, num_classes)

    # Load datasets
    train_dataset = load_dataset(train_file_pattern, batch_size)
    val_dataset = load_dataset(val_file_pattern, batch_size)
    test_dataset = load_dataset(test_file_pattern, batch_size)

    # Train the model
    history = train_model(model, train_dataset, val_dataset, epochs=5)

    # Evaluate the model
    evaluate_model(model, test_dataset)
```

### Explanation:
1. **Model Architecture**: A simple CNN with three convolutional blocks followed by global average pooling and a dense layer with sigmoid activation for multi-label classification.
2. **Data Loading**: Assumes data is stored in TFRecord format with `mel_spectrogram` and `labels` fields. Adjust the parsing logic if your data format differs.
3. **Training Loop**: Compiles the model with binary cross-entropy loss and AUC metric, fits the model on training data while validating on validation data.
4. **Evaluation Function**: Evaluates the trained model on a test dataset.

This setup allows for rapid iterations due to its simplicity and efficiency, making it suitable for the constraints of the competition.
