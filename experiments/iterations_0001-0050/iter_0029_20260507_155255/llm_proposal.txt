```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
import math
from tqdm import tqdm

# Data loading setup
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=32)

# Define the CNN model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdCLEFModel, self).__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=(5, 5), stride=(1, 1), padding=(2, 2))
        self.bn1 = nn.BatchNorm2d(32)
        self.pool = nn.MaxPool2d(kernel_size=(2, 2), stride=(2, 2))
        
        self.conv2 = nn.Conv2d(32, 64, kernel_size=(5, 5), stride=(1, 1), padding=(2, 2))
        self.bn2 = nn.BatchNorm2d(64)
        
        self.adaptive_pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc1 = nn.Linear(64 * 1 * 1, num_classes)

    def forward(self, x):
        x = self.pool(F.relu(self.bn1(self.conv1(x))))
        x = self.pool(F.relu(self.bn2(self.conv2(x))))
        x = self.adaptive_pool(x)
        x = x.view(x.size(0), -1)  # Flatten the tensor
        x = torch.sigmoid(self.fc1(x))
        return x

# Initialize model, loss function, and optimizer
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = BirdCLEFModel(num_classes=234).to(device)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Training loop with limited data for quick iteration
num_epochs = 3
total_samples = 5000
samples_counter = 0

for epoch in range(num_epochs):
    model.train()
    running_loss = 0.0
    with tqdm(train_loader, desc=f'Epoch {epoch+1}/{num_epochs}') as t:
        for inputs, labels in t:
            if samples_counter >= total_samples:
                break
            
            # Move data to device
            inputs, labels = inputs.to(device), labels.to(device)
            
            # Zero the parameter gradients
            optimizer.zero_grad()
            
            # Forward pass
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            
            # Backward pass and optimize
            loss.backward()
            optimizer.step()
            
            running_loss += loss.item() * inputs.size(0)
            samples_counter += inputs.size(0)
            t.set_postfix(loss=running_loss / samples_counter)

    epoch_loss = running_loss / samples_counter
    print(f'Epoch {epoch+1} Loss: {epoch_loss:.4f}')
    
    # Clear cache to prevent OOM
    torch.cuda.empty_cache()

# Calculate final training loss and model size
model.eval()
final_train_loss = epoch_loss
num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)

# Save the trained model
torch.save(model.state_dict(), 'model.pt')

# Prepare metrics dictionary
metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": 0.0,  # Placeholder; compute AUC if possible
    "num_params": num_params,
    "epochs_trained": num_epochs,
    "batch_size": train_loader.batch_size
}

# Write metrics to JSON file and print
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS:", json.dumps(metrics))
```

1. **Model Architecture**: The model consists of two convolutional layers followed by batch normalization, ReLU activation, and max-pooling layers. An adaptive average pooling layer is used to flatten the features robustly before feeding them into a fully connected layer with a sigmoid output for multi-label classification.

2. **Training Loop**:
   - The loop iterates over the training data loader.
   - It uses a limited number of samples (5000) and epochs (3) as per the constraints to ensure quick iterations.
   - The model's parameters are updated using the Adam optimizer with binary cross-entropy loss (`BCELoss`).

3. **Metrics Collection**:
   - After training, the final training loss is recorded.
   - Model size in terms of the number of trainable parameters is computed.
   - Metrics are saved to `metrics.json` and printed to stdout.

4. **Data Handling**: The data loader is assumed to be correctly implemented as per the provided code snippet.
