```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Data ingestion
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=32)

# Define a simple CNN architecture for the task
class BirdClassifier(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdClassifier, self).__init__()
        self.conv1 = nn.Conv2d(1, 64, kernel_size=(3, 3), stride=(1, 1), padding=(1, 1))
        self.bn1 = nn.BatchNorm2d(64)
        self.relu = nn.ReLU()
        
        self.conv2 = nn.Conv2d(64, 128, kernel_size=(3, 3), stride=(1, 1), padding=(1, 1))
        self.bn2 = nn.BatchNorm2d(128)
        
        self.pool = nn.MaxPool2d(kernel_size=(2, 2), stride=(2, 2))
        self.avgpool = nn.AdaptiveAvgPool2d(output_size=(1, 1))
        
        self.fc1 = nn.Linear(128, num_classes)

    def forward(self, x):
        x = self.conv1(x)
        x = self.bn1(x)
        x = self.relu(x)
        x = self.pool(x)
        
        x = self.conv2(x)
        x = self.bn2(x)
        x = self.relu(x)
        x = self.pool(x)
        
        x = self.avgpool(x)
        x = torch.flatten(x, 1)  # Flatten all dimensions except batch
        x = self.fc1(x)
        x = torch.sigmoid(x)  # Sigmoid activation for multi-label classification
        return x

# Initialize model, loss function, and optimizer
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model = BirdClassifier(num_classes=234).to(device)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Training loop with limited data for quick iteration
num_epochs = 3
subset_size = 5000
train_loss_values = []

for epoch in range(num_epochs):
    model.train()
    running_loss = 0.0
    processed_samples = 0
    
    for inputs, labels in train_loader:
        if processed_samples >= subset_size:
            break
        
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item() * inputs.size(0)
        processed_samples += inputs.size(0)
    
    epoch_loss = running_loss / processed_samples
    train_loss_values.append(epoch_loss)
    print(f'Epoch {epoch+1}/{num_epochs}, Loss: {epoch_loss:.4f}')
    
    torch.cuda.empty_cache()

# Evaluation code to capture AUC metric
import numpy as np
from sklearn.metrics import roc_auc_score

model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

# Calculate macro AUC
final_auc = float(roc_auc_score(all_labels, all_preds, average='macro'))

# Save metrics and model
metrics_dict = {
    "final_train_loss": train_loss_values[-1],
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": num_epochs,
    "batch_size": 32
}

with open('metrics.json', 'w') as f:
    json.dump(metrics_dict, f)

print("METRICS: " + json.dumps(metrics_dict))

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: 
   - The model consists of two convolutional layers followed by batch normalization and ReLU activation.
   - Max pooling is used to reduce spatial dimensions, and an adaptive average pooling layer ensures robust flattening irrespective of input size variations.
   - A single fully connected layer maps the flattened features to the number of classes (234 species).
   - Sigmoid activation is applied at the output for multi-label classification.

2. **Training Loop**:
   - The model is trained on a subset of 5000 samples to ensure quick iterations.
   - BCELoss is used as it's appropriate for binary cross-entropy in multi-label scenarios.
   - Adam optimizer with a learning rate of 0.001 is employed.

3. **Evaluation**:
   - After training, the model is evaluated on the same dataset subset using macro AUC to capture performance across all species labels.
   - Metrics are saved to `metrics.json` and printed as a single line JSON string for easy comparison.
   - The trained model state dictionary is saved as `model.pt` for later use.
