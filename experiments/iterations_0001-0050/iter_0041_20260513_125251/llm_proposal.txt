```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define the model architecture based on iteration number
iteration = 1  # This is your current iteration number

DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
batch_size = 16 if iteration % 3 == 1 else (8 if iteration % 3 == 0 else 32)
lr = 0.0001 if iteration % 3 == 1 else (0.001 if iteration % 3 == 0 else 0.00001)
num_conv_layers = 3 if iteration % 3 == 1 else (2 if iteration % 3 == 0 else 4)
dropout_rate = 0.2
use_batchnorm_everywhere = True

train_loader, val_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=3, dropout_rate=0.2, use_batchnorm_everywhere=True):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for i in range(num_conv_layers):
            out_channels = 64 * (2 ** i)
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, stride=1, padding=1))
            if use_batchnorm_everywhere:
                layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
            layers.append(nn.MaxPool2d(kernel_size=2, stride=2))
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc = nn.Linear(in_channels, num_classes)
        self.dropout = nn.Dropout(dropout_rate)

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).squeeze()
        x = self.dropout(x)
        x = self.fc(x)
        return torch.sigmoid(x)

model = BirdCLEFModel(num_conv_layers=num_conv_layers, dropout_rate=dropout_rate, use_batchnorm_everywhere=use_batchnorm_everywhere)
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)

criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.001)  # L2 regularization

# Training loop
epochs_trained = 2
training_samples = 5000
eval_samples = len(val_loader.dataset)
num_params = sum(p.numel() for p in model.parameters() if p.requires_grad)

for epoch in range(epochs_trained):
    model.train()
    running_loss = 0.0
    for i, (inputs, labels) in enumerate(train_loader):
        if i * batch_size >= training_samples:
            break
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item() * inputs.size(0)
    
    epoch_loss = running_loss / training_samples
    print(f"Epoch {epoch + 1}/{epochs_trained} - Training Loss: {epoch_loss:.4f}")
    torch.cuda.empty_cache()

# Evaluation loop
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.array(all_preds)
all_labels = np.array(all_labels)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON
metrics = {
    "final_train_loss": epoch_loss.item(),
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": epochs_trained,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": eval_samples
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: The `BirdCLEFModel` class defines a CNN with a variable number of convolutional layers (2, 3, or 4 based on the iteration). Each layer includes optional batch normalization and max pooling. The final feature map is flattened using `AdaptiveAvgPool2d` to ensure robust dimension handling.

2. **Training Loop**: The training loop iterates over the training dataset for a limited number of epochs (2 in this case) and uses a subset of the first 5000 samples for quick feedback. It includes standard steps like zeroing gradients, computing loss, backpropagation, and optimization.

3. **Evaluation Loop**: After training, the model is evaluated on the validation set. Predictions are collected and used to compute the weighted AUC score using `roc_auc_score` from scikit-learn.

4. **Metrics Capture & Model Saving**:
   - The final training loss and validation AUC are captured.
   - Metrics are saved in a JSON file, including details about the model architecture and training process.
   - The trained model weights are saved to enable reuse or further experimentation.

This setup ensures that each iteration is fast and can be used systematically to explore different hyperparameters efficiently.
