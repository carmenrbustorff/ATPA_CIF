```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define the CNN model architecture based on iteration number for systematic exploration
def create_model(iteration):
    if iteration % 3 == 0:
        num_conv_layers = 2
        batch_size = 8
        lr = 0.001
        dropout_rate = 0.0
    elif iteration % 3 == 1:
        num_conv_layers = 3
        batch_size = 16
        lr = 0.0001
        dropout_rate = 0.2
    else:
        num_conv_layers = 4
        batch_size = 32
        lr = 0.00001
        dropout_rate = 0.5

    conv_layers = []
    in_channels = 1
    out_channels = 32
    for _ in range(num_conv_layers):
        conv_layers.extend([
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(),
        ])
        in_channels = out_channels
        out_channels *= 2

    model = nn.Sequential(
        *conv_layers,
        nn.AdaptiveAvgPool2d((1, 1)),
        nn.Flatten(),
        nn.Linear(in_channels, 128),
        nn.ReLU(),
        nn.Dropout(dropout_rate),
        nn.Linear(128, 234)
    )
    
    return model, batch_size, lr

# Define the training loop
def train_model(model, train_loader, num_epochs=2):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model.to(device)
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.0001)

    model.train()
    total_steps = len(train_loader)
    all_labels = []
    all_preds = []

    for epoch in range(num_epochs):
        running_loss = 0.0
        for i, (inputs, labels) in enumerate(train_loader):
            inputs, labels = inputs.to(device), labels.float().to(device)

            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

            running_loss += loss.item()
            
            # Collect predictions and labels for AUC computation
            all_preds.append(torch.sigmoid(outputs).cpu().detach().numpy())
            all_labels.append(labels.cpu().detach().numpy())

            if (i+1) % 10 == 0:
                print(f'Epoch [{epoch+1}/{num_epochs}], Step [{i+1}/{total_steps}], Loss: {running_loss / (i+1):.4f}')
        
        torch.cuda.empty_cache()
    
    return model, running_loss / total_steps

# Define the evaluation function
def evaluate_model(model, dataloader):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    model.to(device)
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in tqdm(dataloader):
            inputs, labels = inputs.to(device), labels.float().to(device)
            outputs = model(inputs)
            
            # Collect predictions and labels
            all_preds.append(torch.sigmoid(outputs).cpu().numpy())
            all_labels.append(labels.cpu().numpy())

    from sklearn.metrics import roc_auc_score
    import numpy as np

    all_preds = np.concatenate(all_preds)
    all_labels = np.concatenate(all_labels)

    final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    return final_auc

# Set iteration number
iteration = 1

# Create model and get data loader
model, batch_size, lr = create_model(iteration)
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Train the model
trained_model, final_train_loss = train_model(model, train_loader)

# Evaluate the model on the same loader in eval mode
eval_samples = len(train_loader.dataset) // 2  # Assuming half for validation
train_loader.dataset.set_eval_mode(True)
final_auc = evaluate_model(trained_model, train_loader)

# Save metrics to JSON
metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": 2,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset) - eval_samples,
    "eval_samples": eval_samples
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

# Save the model checkpoint
torch.save(trained_model.state_dict(), 'model.pt')

print("METRICS:", json.dumps(metrics))
```

### Explanation:

1. **Model Architecture**: The architecture is dynamically created based on the iteration number to explore different configurations systematically. For `iteration % 3 == 0`, it uses two convolutional layers; for `iteration % 3 == 1`, three convolutional layers; and for `iteration % 3 == 2`, four convolutional layers.

2. **Training Loop**: The training loop iterates over the dataset, updating the model parameters using Adam optimizer with a learning rate that changes based on the iteration number. It uses binary cross-entropy loss as the criterion since this is a multi-label classification problem.

3. **Evaluation**: After training, the model's performance is evaluated using AUC (Area Under Curve) on half of the dataset used for validation. The dataset loader is switched to evaluation mode by calling `set_eval_mode(True)`.

4. **Metrics and Checkpointing**: Training loss, validation AUC, number of parameters, and other relevant metrics are saved in a JSON file named `metrics.json`. Model weights are also saved as `model.pt` for potential reuse or further training.

5. **Systematic Exploration**: By varying the architecture depth, learning rate, batch size, and dropout rate systematically across iterations, this approach aims to find an effective model configuration quickly.

This setup allows for rapid experimentation with different hyperparameters while adhering to the constraints of the competition.
