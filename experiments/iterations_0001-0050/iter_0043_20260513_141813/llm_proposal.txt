```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants and Data Loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
iteration = 1  # Assuming this is the first iteration

# Hyperparameters based on iteration number
batch_size_options = [8, 16, 32]
learning_rate_options = [0.001, 0.0001, 0.00001]
conv_layer_options = [2, 3, 4]
dropout_options = [0.0, 0.2, 0.5]

batch_size = batch_size_options[iteration % 3]
lr = learning_rate_options[iteration % 3]
num_conv_layers = conv_layer_options[iteration % 3]
dropout_rate = dropout_options[iteration % 3]

train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Model Definition
class BirdClassifier(nn.Module):
    def __init__(self, num_classes=234, num_conv_layers=3, dropout_rate=0.5):
        super(BirdClassifier, self).__init__()
        layers = []
        in_channels = 1
        out_channels = 32
        
        for _ in range(num_conv_layers):
            layers.extend([
                nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
                nn.BatchNorm2d(out_channels),
                nn.ReLU(),
                nn.MaxPool2d(kernel_size=2, stride=2)
            ])
            in_channels = out_channels
            out_channels *= 2
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.dropout = nn.Dropout(dropout_rate)
        self.classifier = nn.Sequential(
            nn.Linear(out_channels, 128),
            nn.ReLU(),
            nn.Linear(128, num_classes)  # NO BatchNorm1d!
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        x = self.dropout(x)
        return torch.sigmoid(self.classifier(x))

# Model Initialization
model = BirdClassifier(num_classes=234, num_conv_layers=num_conv_layers, dropout_rate=dropout_rate)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Training Setup
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.0001)  # L2 regularization

# Training Loop
num_epochs = 2
total_training_samples = min(5000, len(train_loader.dataset))
eval_samples = total_training_samples // 4  # Use a subset for validation

model.train()
train_loss_list = []

for epoch in range(num_epochs):
    running_loss = 0.0
    model.train()
    
    for i, (inputs, labels) in enumerate(train_loader):
        if i * batch_size >= total_training_samples:
            break
        
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item() * inputs.size(0)
    
    epoch_loss = running_loss / total_training_samples
    train_loss_list.append(epoch_loss)
    print(f"Epoch {epoch + 1}/{num_epochs}, Loss: {epoch_loss:.4f}")
    
    torch.cuda.empty_cache()

# Evaluation Loop
model.eval()
all_preds, all_labels = [], []

with torch.no_grad():
    for i, (inputs, labels) in enumerate(train_loader):
        if i * batch_size >= eval_samples:
            break
        
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

# Ensure no NaN or Inf values in predictions
all_preds = np.clip(all_preds, 1e-7, 1 - 1e-7)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Validation AUC: {final_auc:.4f}")

# Metrics and Model Saving
num_params = sum(p.numel() for p in model.parameters())
metrics = {
    "final_train_loss": train_loss_list[-1],
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": num_epochs,
    "batch_size": batch_size,
    "training_samples": total_training_samples,
    "eval_samples": eval_samples
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save the model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Imports and Data Loading**: We start with the required imports and load the dataset using `get_dataloader`. The batch size, learning rate, number of convolutional layers, and dropout rate are set based on the iteration number.

2. **Model Architecture**:
   - The model consists of multiple convolutional layers followed by batch normalization and ReLU activation.
   - We use `AdaptiveAvgPool2d` to ensure robust flattening regardless of input dimensions.
   - A dropout layer is added for regularization.
   - The final classifier is a two-layer linear network with ReLU activation.

3. **Training Loop**:
   - We train the model for 2 epochs using the specified hyperparameters.
   - The training loop includes loss computation, backpropagation, and optimizer step.
   - We limit the number of training samples to 5000 for quick iterations.

4. **Evaluation Loop**:
   - We evaluate the model on a subset of the data (eval_samples) using the same dataloader in evaluation mode.
   - Predictions are clipped to avoid numerical instability in `roc_auc_score`.

5. **Metrics Capture and Model Saving**:
   - We compute the validation AUC score.
   - Metrics are saved to a JSON file, including training loss, AUC, number of parameters, and other relevant details.
   - The model's state dictionary is saved for later use or submission.

This setup allows for rapid experimentation with different hyperparameters while adhering to the competition rules.
