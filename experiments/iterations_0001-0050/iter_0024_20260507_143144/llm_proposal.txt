### Model Architecture

We will design a simple CNN with several convolutional layers followed by max-pooling and fully connected (dense) layers. This architecture is inspired by architectures commonly used in audio classification tasks, specifically for mel-spectrogram inputs.

#### Key Components:
- **Convolutional Layers**: To capture local patterns.
- **Max Pooling Layers**: For down-sampling the feature maps.
- **Batch Normalization**: For faster convergence and stability.
- **Dropout Layers**: For regularization.
- **Fully Connected Layers**: To combine features from previous layers.

### PyTorch Implementation

Below is the complete code for the model, training loop, and evaluation. The model will be trained on a GPU if available.

```python
import os
import torch
import torch.nn as nn
import torch.optim as optim
from dataset import get_dataloader

# Set device to GPU if available
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# Define the model architecture
class BirdModel(nn.Module):
    def __init__(self, num_classes=234):
        super(BirdModel, self).__init__()
        
        # Convolutional layers
        self.conv1 = nn.Conv2d(in_channels=1, out_channels=32, kernel_size=(5, 5), stride=(1, 1), padding=2)
        self.bn1 = nn.BatchNorm2d(32)
        self.pool1 = nn.MaxPool2d(kernel_size=(2, 2))
        
        self.conv2 = nn.Conv2d(in_channels=32, out_channels=64, kernel_size=(5, 5), stride=(1, 1), padding=2)
        self.bn2 = nn.BatchNorm2d(64)
        self.pool2 = nn.MaxPool2d(kernel_size=(2, 2))
        
        self.conv3 = nn.Conv2d(in_channels=64, out_channels=128, kernel_size=(5, 5), stride=(1, 1), padding=2)
        self.bn3 = nn.BatchNorm2d(128)
        self.pool3 = nn.MaxPool2d(kernel_size=(2, 2))
        
        # Fully connected layers
        self.fc1 = nn.Linear(in_features=128 * 16 * 16, out_features=512)  # Adjusted based on output size after convolutions
        self.dropout1 = nn.Dropout(0.3)
        self.fc2 = nn.Linear(in_features=512, out_features=num_classes)

    def forward(self, x):
        x = torch.relu(self.pool1(self.bn1(self.conv1(x))))
        x = torch.relu(self.pool2(self.bn2(self.conv2(x))))
        x = torch.relu(self.pool3(self.bn3(self.conv3(x))))
        
        # Flatten the tensor
        x = x.view(-1, 128 * 16 * 16)  # Adjusted based on output size after convolutions
        
        x = torch.relu(self.fc1(x))
        x = self.dropout1(x)
        x = torch.sigmoid(self.fc2(x))  # Sigmoid for multi-label classification
        return x

# Initialize the model, loss function, and optimizer
model = BirdModel(num_classes=234).to(device)
criterion = nn.BCEWithLogitsLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Load data
DATA_DIR = "/home/carme/birdclef-data"
METADATA_FILE = os.path.join(DATA_DIR, "train_metadata.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=32)

# Training loop
num_epochs = 10

for epoch in range(num_epochs):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Zero the parameter gradients
        optimizer.zero_grad()
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        
        # Backward pass and optimize
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
    
    print(f'Epoch [{epoch+1}/{num_epochs}], Loss: {running_loss/len(train_loader):.4f}')

# Simple evaluation function (accuracy)
def evaluate_model(model, data_loader):
    model.eval()
    correct = 0
    total = 0
    
    with torch.no_grad():
        for inputs, labels in data_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            
            outputs = model(inputs)
            predicted = (outputs > 0.5).float()  # Convert to binary
            
            # Count the number of correctly classified samples
            total += labels.size(0)
            correct += torch.sum((predicted == labels).all(dim=1)).item()
    
    accuracy = correct / total
    return accuracy
```

- **Training Loop**: We iterate over epochs, compute the loss using binary cross-entropy with logits, perform backpropagation, and update the weights.
