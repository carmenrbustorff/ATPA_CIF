```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score
import torchaudio.transforms as T

# Define hyperparameters
NUM_SPECIES = 206
max_epochs = 40
early_stopping_patience = 5

# Select hyperparameters based on iteration number
iteration = 3  # Example iteration number, replace with actual iteration logic
batch_size_options = [8, 16, 32]
learning_rate_options = [0.001, 0.0001, 0.00001]
dropout_options = [0.0, 0.2, 0.5]

# Choose hyperparameters
batch_size = batch_size_options[iteration % 3]
lr = learning_rate_options[iteration % 3]
dropout = dropout_options[iteration % 3]
num_conv_layers = iteration % 3 + 2  # 2 to 4 layers

DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=batch_size,
    augment=True,
    val_split=0.2,
)

# Define the CNN model
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, num_conv_layers, dropout_rate):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        
        # Convolutional layers
        for i in range(num_conv_layers):
            out_channels = 32 * (i + 1)
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
            in_channels = out_channels
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Classifier
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, optimizer
model = BirdCLEFModel(NUM_SPECIES, num_conv_layers, dropout)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)

if torch.cuda.is_available():
    model.to('cuda')

# Early stopping setup
best_val_auc = 0.0
patience_counter = 0

# Training loop
for epoch in range(max_epochs):
    model.train()
    running_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        if torch.cuda.is_available():
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        
        # Data augmentation
        freq_mask = T.FrequencyMasking(freq_mask_param=80)
        time_mask = T.TimeMasking(time_mask_param=80)
        inputs = time_mask(freq_mask(inputs))
        
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        running_loss += loss.item()
        
        # Clear cache and reduce batch size if OOM error occurs
        try:
            torch.cuda.empty_cache()
        except RuntimeError as e:
            print("CUDA out of memory. Reducing batch size.")
            batch_size //= 2
            train_loader, val_loader = build_train_val_dataloaders(
                metadata_csv=METADATA_FILE,
                audio_dir=DATA_DIR,
                batch_size=batch_size,
                augment=True,
                val_split=0.2,
            )
            break
    
    # Evaluation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            if torch.cuda.is_available():
                inputs, labels = inputs.to('cuda'), labels.to('cuda')
            
            outputs = model(inputs)
            preds = (outputs > 0.5).float().cpu().numpy()
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())
    
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Epoch {epoch + 1}, Loss: {running_loss / len(train_loader)}, Val AUC: {val_auc}")
    
    # Early stopping
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        patience_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        patience_counter += 1
        if patience_counter >= early_stopping_patience:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": running_loss / len(train_loader),
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Explanation:

1. **Hyperparameter Selection**: The script selects hyperparameters based on the iteration number. This allows for systematic exploration of different configurations.

2. **Model Architecture**: The `BirdCLEFModel` class defines a convolutional neural network with a variable number of convolutional layers and dropout regularization to prevent overfitting.

3. **Training Loop**:
   - Data augmentation is applied using `FrequencyMasking` and `TimeMasking` from `torchaudio.transforms`.
   - The model is trained for a fixed number of epochs, or early stopping occurs if the validation AUC does not improve.

4. **Evaluation**: 
   - After each epoch, the model is evaluated on the validation set.
   - The validation AUC is computed using `roc_auc_score`.

5. **Early Stopping**: If the validation AUC does not improve for a specified number of epochs (`early_stopping_patience`), training stops early.

6. **Metrics and Model Saving**:
   - Metrics are saved to `metrics.json`, including final training loss, best validation AUC, number of parameters, epochs trained, batch size, and sample counts.
   - The model's state dictionary is saved to `model.pt`.

This script provides a framework for rapid experimentation with different hyperparameters and architectures, allowing you to quickly iterate towards better performance.
