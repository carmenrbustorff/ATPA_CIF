Given that this is the 10th iteration, we will:

- **Adjust Learning Rate**: Use a reduced learning rate of 0.0001 for fine-tuning.

**Code Snippet:**
```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define hyperparameters
NUM_SPECIES = 206
EPOCHS = 10
BATCH_SIZE = 32
LR = 0.0001
DROPOUT_RATE = 0.5

class BirdCLEFModel(nn.Module):
    def __init__(self, num_species):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.Conv2d(128, 256, 3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),
        )
        self.pool = nn.MaxPool2d(2, 2)
        self.fc_layers = nn.Sequential(
            nn.Linear(256 * 16 * 16, 512), nn.ReLU(), nn.Dropout(DROPOUT_RATE),
            nn.Linear(512, num_species), nn.Sigmoid()
        )

    def forward(self, x):
        x = self.pool(self.conv_layers(x))
        x = torch.flatten(x, 1)  # flatten all dimensions except batch
        x = self.fc_layers(x)
        return x

# Main training loop
def train_model():
    DATA_DIR = "/mnt/disks/data/birdclef"
    METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
    
    train_loader = get_dataloader(METADATA_FILE, batch_size=BATCH_SIZE, shuffle=True)
    model = BirdCLEFModel(NUM_SPECIES).to('cuda')
    criterion = nn.BCELoss()
    optimizer = optim.Adam(model.parameters(), lr=LR)

    for epoch in range(EPOCHS):
        running_loss = 0.0
        for inputs, labels in tqdm(train_loader):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')

            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            loss.backward()
            optimizer.step()

            running_loss += loss.item()

        print(f'Epoch {epoch+1}, Loss: {running_loss / len(train_loader)}')

    # Validation
    model.eval()
    with torch.no_grad():
        all_preds, all_labels = [], []
        for inputs, labels in tqdm(train_loader):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            preds = (outputs > 0.5).float().cpu().numpy()
            labels = labels.cpu().numpy()

            all_preds.append(preds)
            all_labels.append(labels)

    from sklearn.metrics import roc_auc_score
    final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Validation AUC: {final_auc}")

    # Save metrics and model
    num_params = sum(p.numel() for p in model.parameters())
    metrics = {
        "final_train_loss": running_loss / len(train_loader),
        "final_auc": final_auc,
        "num_params": num_params,
        "epochs_trained": EPOCHS,
        "batch_size": BATCH_SIZE,
        "training_samples": 15000,  # Adjust based on actual training samples
        "eval_samples": 5000      # Adjust based on actual eval samples
    }
    with open('metrics.json', 'w') as f:
        json.dump(metrics, f)
    print("METRICS: " + json.dumps(metrics))
    
    torch.save(model.state_dict(), 'model.pt')

if __name__ == "__main__":
    train_model()
```
