```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Hyperparameters
NUM_EPOCHS = 20
MAX_BATCH_SIZE = 32
LEARNING_RATE = 0.001
DROPOUT_PROB = 0.5
NUM_CONV_LAYERS = 3
WEIGHT_DECAY = 1e-4
PATIENCE = 5

# Data paths
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Load data
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=MAX_BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

NUM_CLASSES = len(train_loader.dataset.classes)
print(f"Number of classes: {NUM_CLASSES}")

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes, num_conv_layers=3, dropout_prob=0.5):
        super(BirdCLEFModel, self).__init__()
        channels = [16, 32, 64, 128][:num_conv_layers]
        layers = []
        
        for i in range(num_conv_layers):
            if i == 0:
                in_channels = 1
            else:
                in_channels = channels[i-1]
            
            out_channels = channels[i]
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 256),
            nn.ReLU(),
            nn.Dropout(dropout_prob),
            nn.Linear(256, num_classes)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(NUM_CLASSES, NUM_CONV_LAYERS, DROPOUT_PROB)

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY)

# Early stopping
best_auc = 0.0
patience_counter = 0

# Training loop
for epoch in range(NUM_EPOCHS):
    model.train()
    train_loss = 0.0
    batch_count = 0
    
    for inputs, labels in tqdm(train_loader):
        if batch_count >= 10:  # Limit to N batches for fast feedback
            break
        
        inputs = inputs.to('cuda')
        labels = labels.float().to('cuda')
        
        try:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            
            train_loss += loss.item()
            batch_count += 1
        except RuntimeError as e:
            if 'CUDA out of memory' in str(e):
                print("OOM error, skipping this batch.")
                torch.cuda.empty_cache()
            else:
                raise e

    # Validation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs = inputs.to('cuda')
            labels = labels.float().to('cuda')
            
            outputs = model(inputs)
            preds = (outputs > 0.5).float()
            
            all_preds.append(preds.cpu())
            all_labels.append(labels.cpu())

    all_preds = torch.cat(all_preds, dim=0)
    all_labels = torch.cat(all_labels, dim=0)

    # Calculate AUC
    from sklearn.metrics import roc_auc_score
    auc = roc_auc_score(all_labels.numpy(), all_preds.numpy(), average='weighted')
    
    print(f"Epoch {epoch+1}, Train Loss: {train_loss/batch_count:.4f}, Val AUC: {auc:.4f}")
    
    # Early stopping check
    if auc > best_auc:
        best_auc = auc
        patience_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        patience_counter += 1
        if patience_counter >= PATIENCE:
            print(f"Early stopping at epoch {epoch+1}")
            break

# Save metrics
metrics = {
    "final_train_loss": train_loss / batch_count,
    "final_auc": best_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": NUM_EPOCHS,
    "batch_size": MAX_BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```
