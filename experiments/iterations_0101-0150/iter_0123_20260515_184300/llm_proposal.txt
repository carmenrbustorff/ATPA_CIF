```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Constants
NUM_SPECIES = 206
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Model Architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, conv_layers=3, dropout_rate=0.5):
        super(BirdCLEFModel, self).__init__()
        
        # Define convolutional layers dynamically based on the number specified
        conv_blocks = []
        in_channels = 1
        out_channels = [32, 64, 128, 256][:conv_layers]
        for i in range(conv_layers):
            conv_blocks.append(nn.Conv2d(in_channels, out_channels[i], kernel_size=3, padding=1))
            conv_blocks.append(nn.BatchNorm2d(out_channels[i]))
            conv_blocks.append(nn.ReLU())
            in_channels = out_channels[i]
        
        self.conv_layers = nn.Sequential(*conv_blocks)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Classifier
        classifier_layers = []
        classifier_layers.append(nn.Linear(in_channels, 128))
        classifier_layers.append(nn.ReLU())
        classifier_layers.append(nn.Dropout(dropout_rate))
        classifier_layers.append(nn.Linear(128, num_species))
        
        self.classifier = nn.Sequential(*classifier_layers)
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data Loading
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=32,
    augment=True,
    val_split=0.2,
)

# Hyperparameters for the current iteration (example)
iteration = 4
batch_size_options = {0: 8, 1: 16, 2: 32}
learning_rate_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
conv_layers_options = {0: 2, 1: 3, 2: 4}
dropout_rates = [0.0, 0.2, 0.5]

batch_size = batch_size_options[iteration % 3]
learning_rate = learning_rate_options[iteration % 3]
num_conv_layers = conv_layers_options[iteration % 3]
dropout_rate = dropout_rates[iteration % 3]

# Instantiate model, optimizer, and loss function
model = BirdCLEFModel(num_species=NUM_SPECIES, conv_layers=num_conv_layers, dropout_rate=dropout_rate)
model.to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate, weight_decay=1e-4)

# Training loop
max_epochs = 20
early_stopping_patience = 5
best_auc = 0.0
epochs_without_improvement = 0

for epoch in range(max_epochs):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        try:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            loss.backward()
            optimizer.step()
            running_loss += loss.item()
            
        except RuntimeError as e:
            if 'out of memory' in str(e):
                print("OOM error caught. Running empty cache and reducing batch size.")
                torch.cuda.empty_cache()
                batch_size //= 2
                train_loader.batch_size = batch_size
                continue
        
        torch.cuda.empty_cache()  # Clear cached allocations
    
    epoch_loss = running_loss / len(train_loader)
    
    # Validation
    model.eval()
    all_preds, all_labels = [], []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    
    # Calculate AUC
    from sklearn.metrics import roc_auc_score
    auc_score = roc_auc_score(all_labels, all_preds, average='weighted')
    
    print(f"Epoch {epoch+1}/{max_epochs}, Train Loss: {epoch_loss:.4f}, Val AUC: {auc_score:.4f}")
    
    # Check for early stopping and save best model
    if auc_score > best_auc:
        best_auc = auc_score
        epochs_without_improvement = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        epochs_without_improvement += 1
        if epochs_without_improvement >= early_stopping_patience:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": epoch_loss,
    "final_auc": best_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + str(metrics))
```

This script defines a new model architecture with dynamic convolutional layer depth and dropout rate based on the iteration number. It includes a training loop with OOM handling and early stopping, as well as evaluation and saving of metrics and model checkpoints. The script is designed to be run iteratively for rapid experimentation and hyperparameter tuning.
