```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants
NUM_SPECIES = 206
MAX_EPOCHS = 40
EARLY_STOPPING_PATIENCE = 5

# Hyperparameters based on iteration number (example for Iteration 1)
iteration_number = 1
batch_size = 16 if iteration_number % 3 == 1 else 8 if iteration_number % 3 == 2 else 32
lr = 0.0001 if iteration_number % 3 == 1 else 0.001 if iteration_number % 3 == 2 else 0.00001
num_conv_layers = 3 if iteration_number % 3 == 1 else 4 if iteration_number % 3 == 2 else 2
dropout_rate = 0.2

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size, augment=True)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, num_conv_layers=3, dropout_rate=0.2):
        super(BirdCLEFModel, self).__init__()
        channels = [16, 32, 64, 128][:num_conv_layers]
        
        conv_layers = []
        for i in range(num_conv_layers):
            if i == 0:
                conv_layers.append(nn.Conv2d(1, channels[i], 3, padding=1))
            else:
                conv_layers.append(nn.Conv2d(channels[i-1], channels[i], 3, padding=1))
            conv_layers.append(nn.BatchNorm2d(channels[i]))
            conv_layers.append(nn.ReLU())
        
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 128), nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_species=NUM_SPECIES, num_conv_layers=num_conv_layers, dropout_rate=dropout_rate)
model.to('cuda')

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)

# Training loop
best_auc = 0.0
early_stopping_counter = 0

for epoch in range(MAX_EPOCHS):
    model.train()
    train_loss = 0.0
    
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}"):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_loss += loss.item()
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
    
    avg_train_loss = train_loss / len(train_loader)
    print(f"Epoch {epoch+1}, Train Loss: {avg_train_loss:.4f}")
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in tqdm(train_loader, desc="Validation"):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.append(outputs.cpu().numpy())
            all_labels.append(labels.cpu().numpy())
    
    from sklearn.metrics import roc_auc_score
    all_preds = np.concatenate(all_preds)
    all_labels = np.concatenate(all_labels)
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Validation AUC: {val_auc:.4f}")
    
    # Early stopping
    if val_auc > best_auc:
        best_auc = val_auc
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        early_stopping_counter += 1
        if early_stopping_counter >= EARLY_STOPPING_PATIENCE:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": best_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(train_loader.dataset)  # Assuming no separate validation set
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Explanation

1. **Model Architecture**: The `BirdCLEFModel` class is defined with a flexible number of convolutional layers and dropout rate based on the iteration number. It uses ReLU activations and Batch Normalization for stability.

2. **Training Loop**: The loop iterates over epochs, updating weights using Adam optimizer with weight decay to prevent overfitting. Losses are averaged per epoch.

3. **Validation**: The same dataloader is used for validation in `eval` mode, where predictions are collected and AUC is computed.

4. **Early Stopping**: If the validation AUC doesn't improve for a set number of epochs (`EARLY_STOPPING_PATIENCE`), training stops early to save time.

5. **Metrics & Model Saving**: Training loss, best AUC, number of parameters, and other metadata are saved in `metrics.json`. The model state is also saved for future use or Kaggle submission.

This setup allows rapid experimentation by varying hyperparameters systematically across iterations.
