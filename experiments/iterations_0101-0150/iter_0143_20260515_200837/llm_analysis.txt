import os
import torch
import pandas as pd
from torch.utils.data import DataLoader
from sklearn.metrics import roc_auc_score
import json

# Define paths
DATA_DIR = "/home/carme/birdclef-data"
METADATA_FILE = os.path.join(DATA_DIR, "metadata.csv")

# Ensure the metadata file exists
if not os.path.exists(METADATA_FILE):
    raise FileNotFoundError(f"Metadata file {METADATA_FILE} does not exist.")

# Load dataset and create dataloaders
def build_train_val_dataloaders(metadata_csv, audio_dir, augment=True, val_split=0.2):
    # Assume BirdCLEFDataset is defined in data_loader.py
    from data_loader import BirdCLEFDataset

    train_dataset = BirdCLEFDataset(metadata_csv=metadata_csv, audio_dir=audio_dir, augment=augment)
    val_dataset = BirdCLEFDataset(metadata_csv=metadata_csv, audio_dir=audio_dir, augment=False)

    # Split dataset into train and validation
    train_size = int(len(train_dataset) * (1 - val_split))
    val_size = len(train_dataset) - train_size

    train_dataset, _ = torch.utils.data.random_split(train_dataset, [train_size, val_size])

    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
    val_loader = DataLoader(val_dataset, batch_size=batch_size, shuffle=False)

    return train_loader, val_loader

# Define CNN architecture
def create_model(num_conv_layers=3, dropout_rate=0.2):
    layers = [
        torch.nn.Conv2d(1, 32, kernel_size=3, stride=1, padding=1),
        torch.nn.BatchNorm2d(32),
        torch.nn.ReLU(),
        torch.nn.MaxPool2d(kernel_size=2, stride=2)
    ]

    for _ in range(num_conv_layers - 1):
        layers.extend([
            torch.nn.Conv2d(32, 32, kernel_size=3, stride=1, padding=1),
            torch.nn.BatchNorm2d(32),
            torch.nn.ReLU(),
            torch.nn.MaxPool2d(kernel_size=2, stride=2)
        ])

    layers.append(torch.nn.AdaptiveAvgPool2d((1, 1)))
    layers.append(torch.nn.Flatten())
    layers.append(torch.nn.Linear(32, num_species))
    layers.append(torch.nn.Sigmoid())

    return torch.nn.Sequential(*layers)

# Training function
def train_model(iteration_number, DATA_DIR, METADATA_FILE):
    # Hyperparameters based on iteration number
    batch_size = [16, 32, 8][iteration_number % 3]
    learning_rate = [0.0001, 0.00001, 0.001][iteration_number % 3]
    num_conv_layers = [3, 4, 2][iteration_number % 3]
    dropout_rate = [0.2, 0.5, 0.0][iteration_number % 3]

    # Load data
    train_loader, val_loader = build_train_val_dataloaders(METADATA_FILE, DATA_DIR)

    # Initialize model
    num_species = 207  # Assuming 207 unique species
    model = create_model(num_conv_layers=num_conv_layers, dropout_rate=dropout_rate)
    model.to('cuda')

    # Loss and optimizer
    criterion = torch.nn.BCELoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=learning_rate)

    # Training loop
    for epoch in range(20):
        model.train()
        running_loss = 0.0
        for inputs, labels in train_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

            running_loss += loss.item() * inputs.size(0)

        epoch_loss = running_loss / len(train_loader.dataset)
        print(f"Epoch {epoch+1}, Loss: {epoch_loss:.4f}")

    # Evaluation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())

    auc_score = roc_auc_score(all_labels, all_preds, average='weighted')

    # Save metrics
    metrics = {
        "final_train_loss": epoch_loss,
        "final_auc": auc_score,
        "num_params": sum(p.numel() for p in model.parameters()),
        "epochs_trained": 20,
        "batch_size": batch_size,
        "training_samples": len(train_loader.dataset),
        "eval_samples": len(val_loader.dataset)
    }

    with open('metrics.json', 'w') as f:
        json.dump(metrics, f)

    print("METRICS: " + json.dumps(metrics))

    # Save model
    torch.save(model.state_dict(), 'model.pt')

# Run training
train_model(iteration_number=143, DATA_DIR=DATA_DIR, METADATA_FILE=METADATA_FILE)
