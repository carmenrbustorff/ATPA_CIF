```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score
import torchaudio.transforms as T

# Define hyperparameters based on the iteration number (for this example, let's assume iteration 4)
iteration = 4
batch_size_options = {0: 8, 1: 16, 2: 32}
lr_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
depth_options = {0: 2, 1: 3, 2: 4}
dropout_options = [0.0, 0.2, 0.5]
weight_decay_options = [0, 0.0001, 0.001]

# Select hyperparameters based on the iteration number
batch_size = batch_size_options[iteration % 3]
lr = lr_options[iteration % 3]
depth = depth_options[iteration % 3]
dropout_rate = dropout_options[(iteration // 3) % len(dropout_options)]
weight_decay = weight_decay_options[(iteration // 9) % len(weight_decay_options)]

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=batch_size,
    augment=True,
    val_split=0.2,
)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, depth, dropout_rate):
        super(BirdCLEFModel, self).__init__()
        channels = [32, 64, 128, 256][:depth]  # Adjust based on depth
        layers = []
        for i in range(len(channels) - 1):
            layers.append(nn.Conv2d(channels[i], channels[i + 1], kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(channels[i + 1]))
            layers.append(nn.ReLU())
        self.conv_layers = nn.Sequential(*layers)
        
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, and optimizer
NUM_SPECIES = 206
model = BirdCLEFModel(NUM_SPECIES, depth, dropout_rate)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)

# Move model to GPU if available
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)

# Training loop
max_epochs = 40
early_stopping_patience = 5
best_val_auc = 0.0
no_improvement_count = 0

for epoch in range(max_epochs):
    model.train()
    total_loss = 0.0
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch + 1}"):
        try:
            inputs, labels = inputs.to(device), labels.to(device)
            
            # Apply SpecAugment
            freq_mask = T.FrequencyMasking(freq_mask_param=80)
            time_mask = T.TimeMasking(time_mask_param=80)
            inputs = freq_mask(time_mask(inputs))
            
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
        except RuntimeError as e:
            print(f"RuntimeError: {e}")
            torch.cuda.empty_cache()

    avg_train_loss = total_loss / len(train_loader)
    print(f"Epoch {epoch + 1}, Train Loss: {avg_train_loss:.4f}")

    # Validation
    model.eval()
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            preds = (outputs >= 0.5).float().cpu().numpy()
            all_preds.extend(preds)
            all_labels.extend(labels.cpu().numpy())

    # Compute AUC
    try:
        val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    except ValueError as e:
        print(f"ValueError: {e}")
        val_auc = 0.0

    print(f"Validation AUC: {val_auc:.4f}")

    # Early stopping
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        no_improvement_count = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        no_improvement_count += 1
        if no_improvement_count >= early_stopping_patience:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": avg_train_loss,
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Key Points:
1. **Hyperparameter Selection**: The hyperparameters are selected based on the iteration number to systematically explore different configurations.
2. **SpecAugment**: Frequency and time masking are applied during training to improve robustness.
3. **Early Stopping**: The training loop includes early stopping to prevent overfitting.
4. **Metrics and Checkpointing**: Training loss, validation AUC, and other metrics are saved to a JSON file, and the model's weights are saved for later use.

This approach allows for rapid experimentation and fine-tuning of the model architecture and hyperparameters.
