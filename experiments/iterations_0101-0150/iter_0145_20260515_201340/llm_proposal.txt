```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Define the CNN model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=206):
        super(BirdCLEFModel, self).__init__()
        # Convolutional layers with BatchNorm and ReLU activation
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
        )
        # Adaptive average pooling for robust dimension handling
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        # Classifier with Linear layers and Dropout for regularization
        self.classifier = nn.Sequential(
            nn.Linear(64, 128), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Hyperparameters
NUM_EPOCHS = 20
BATCH_SIZE = 32
LEARNING_RATE = 0.001
NUM_CLASSES = 206

# Data loading and augmentation
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

# Initialize model, loss function, and optimizer
model = BirdCLEFModel(num_species=NUM_CLASSES).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-4)

# Training loop
for epoch in range(NUM_EPOCHS):
    model.train()
    running_loss = 0.0
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        # Zero the parameter gradients
        optimizer.zero_grad()

        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())

        # Backward pass and optimization
        loss.backward()
        optimizer.step()

        # Update running loss
        running_loss += loss.item()

        # Print batch loss
        if batch_idx % 10 == 9:  # print every 10 mini-batches
            print(f'Epoch [{epoch+1}/{NUM_EPOCHS}], Step [{batch_idx+1}/{len(train_loader)}], Loss: {running_loss/10:.4f}')
            running_loss = 0.0

    # Validate model on the validation set
    model.eval()
    all_preds = []
    all_labels = []
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())

    # Calculate AUC
    from sklearn.metrics import roc_auc_score
    auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f'Validation AUC: {auc:.4f}')

# Save metrics and model checkpoint
metrics = {
    "final_train_loss": running_loss / len(train_loader),
    "final_auc": auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": NUM_EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
torch.save(model.state_dict(), 'model.pt')
```

This script defines a CNN model for the BirdCLEF task, includes a training loop with data loading, and evaluates the model on a validation set. It also saves metrics and the trained model checkpoint as required. The architecture is designed to be flexible for rapid experimentation by varying hyperparameters such as learning rate, batch size, number of convolutional layers, and dropout rate.
