```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
import torchaudio.transforms as T

# Constants
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_SPECIES = 206
MAX_EPOCHS = 40
EARLY_STOPPING_PATIENCE = 5

# Hyperparameters based on iteration number (example for Iteration 1)
iteration = 1
batch_size_options = {0: 8, 1: 16, 2: 32}
learning_rate_options = {0: 0.001, 1: 0.0001, 2: 0.00001}
architecture_depth_options = {0: 2, 1: 3, 2: 4}
dropout_options = {0: 0.0, 1: 0.2, 2: 0.5}

batch_size = batch_size_options[iteration % 3]
lr = learning_rate_options[iteration % 3]
architecture_depth = architecture_depth_options[iteration % 3]
dropout = dropout_options[iteration % 3]

# Data loading
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=batch_size,
    augment=True,
    val_split=0.2,
)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, architecture_depth, dropout):
        super(BirdCLEFModel, self).__init__()
        layers = []
        in_channels = 1
        for _ in range(architecture_depth):
            out_channels = 32 * (architecture_depth + 1)
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_species=NUM_SPECIES, architecture_depth=architecture_depth, dropout=dropout)

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)

# Training loop
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)
best_auc = 0.0
early_stopping_counter = 0

for epoch in range(MAX_EPOCHS):
    model.train()
    total_loss = 0.0
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Data augmentation
        freq_mask = T.FrequencyMasking(freq_mask_param=80)
        time_mask = T.TimeMasking(time_mask_param=80)
        inputs = freq_mask(inputs)
        inputs = time_mask(inputs)

        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()

        total_loss += loss.item()

        # Prevent GPU fragmentation
        torch.cuda.empty_cache()

    print(f'Epoch [{epoch+1}/{MAX_EPOCHS}], Loss: {total_loss / len(train_loader):.4f}')

    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            all_preds.append(outputs.cpu().numpy())
            all_labels.append(labels.cpu().numpy())

    from sklearn.metrics import roc_auc_score
    try:
        final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    except ValueError:
        final_auc = 0.0

    print(f'Validation AUC: {final_auc:.4f}')

    # Early stopping
    if final_auc > best_auc:
        best_auc = final_auc
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')
    else:
        early_stopping_counter += 1
        if early_stopping_counter >= EARLY_STOPPING_PATIENCE:
            print("Early stopping triggered.")
            break

# Metrics and model saving
num_params = sum(p.numel() for p in model.parameters())
metrics = {
    "final_train_loss": total_loss / len(train_loader),
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Explanation:
1. **Data Loading**: The `build_train_val_dataloaders` function is used to load the training and validation datasets.
2. **Model Definition**: A custom `BirdCLEFModel` class is defined with a configurable number of convolutional layers, dropout rate, and architecture depth.
3. **Training Loop**: The model is trained using BCELoss and Adam optimizer. Data augmentation is applied during training to improve generalization.
4. **Validation**: After each epoch, the model's performance is evaluated on the validation set, and metrics like AUC are computed.
5. **Early Stopping**: If the validation AUC does not improve for a certain number of epochs (`EARLY_STOPPING_PATIENCE`), training is stopped early to prevent overfitting.
6. **Metrics and Model Saving**: Training loss, final AUC, number of parameters, and other relevant metrics are saved in `metrics.json`, and the model's state dictionary is saved to `model.pt`.

This setup allows for rapid experimentation with different hyperparameters and architectures while maintaining a structured approach to training and evaluation.
