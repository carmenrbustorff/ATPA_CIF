```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_SPECIES = 206  # Number of unique bird species

# Hyperparameters for iteration 1 (1 % 3 == 1)
batch_size = 16
lr = 0.0001
conv_layers_count = 3
dropout_rate = 0.2
weight_decay = 0.0001
max_epochs = 40
early_stopping_patience = 5

# Model Definition
class BirdCLEFNet(nn.Module):
    def __init__(self, num_species=NUM_SPECIES):
        super(BirdCLEFNet, self).__init__()
        conv_layers = []
        in_channels = 1
        for i in range(conv_layers_count):
            out_channels = 32 * (i + 1)
            conv_layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            conv_layers.append(nn.BatchNorm2d(out_channels))
            conv_layers.append(nn.ReLU())
            in_channels = out_channels
        self.conv_layers = nn.Sequential(*conv_layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data Loading
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size, augment=True)

# Model and Optimizer
model = BirdCLEFNet(num_species=NUM_SPECIES).cuda()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=weight_decay)

# Training Loop
train_losses = []
best_val_auc = 0.0
epochs_without_improvement = 0

for epoch in range(max_epochs):
    model.train()
    for inputs, labels in train_loader:
        inputs, labels = inputs.cuda(), labels.cuda()
        
        # Data Augmentation (Frequency and Time Masking)
        freq_masking = nn.Sequential(
            torchaudio.transforms.FrequencyMasking(freq_mask_param=80),
            torchaudio.transforms.TimeMasking(time_mask_param=80)
        )
        inputs = freq_masking(inputs)

        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        torch.cuda.empty_cache()  # Clear GPU cache to prevent fragmentation

    # Early Stopping Check
    if epochs_without_improvement >= early_stopping_patience:
        print("Early stopping triggered.")
        break

# Evaluation on the same DataLoader in eval mode
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:  # Using train_loader for validation as per constraints
        inputs, labels = inputs.cuda(), labels.cuda()
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

import numpy as np
from sklearn.metrics import roc_auc_score

all_preds = np.concatenate(all_preds)
all_labels = np.concatenate(all_labels)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Final Validation AUC: {final_auc}")

# Save Metrics and Model
metrics = {
    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": batch_size,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(train_loader.dataset)  # Using same dataset for eval as per constraints
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: 
   - The model uses a series of convolutional layers followed by batch normalization and ReLU activations.
   - `AdaptiveAvgPool2d` is used to handle the flattening, ensuring robustness across different input sizes.
   - A dropout layer is added for regularization.

2. **Data Loading**:
   - We use `get_dataloader` with augmentation enabled (`augment=True`) to provide varied training inputs.

3. **Training Loop**:
   - The model is trained using Adam optimizer with the specified learning rate and weight decay.
   - Frequency and time masking are applied as data augmentation techniques within the loop.
   - GPU cache is cleared after each iteration to prevent memory fragmentation.

4. **Evaluation**:
   - The same `train_loader` is used for evaluation as per constraints, moving it into eval mode.
   - AUC is calculated using `roc_auc_score` with a weighted average to handle multi-label classification.

5. **Saving Metrics and Model**:
   - Metrics are saved in JSON format, including training loss, validation AUC, number of parameters, epochs trained, batch size, and sample counts.
   - The model's state dictionary is saved for later use or deployment.
