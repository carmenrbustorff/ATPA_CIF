```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Constants
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_CLASSES = 206
MAX_EPOCHS = 40
EARLY_STOPPING_PATIENCE = 5

def load_data():
    train_loader, val_loader = build_train_val_dataloaders(
        metadata_csv=METADATA_FILE,
        audio_dir=DATA_DIR,
        batch_size=32,
        augment=True,
        val_split=0.2,
    )
    return train_loader, val_loader

class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes, conv_layers):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential()
        for i in range(len(conv_layers) - 1):
            self.conv_layers.add_module(f'conv_{i}', nn.Conv2d(conv_layers[i], conv_layers[i+1], kernel_size=3, padding=1))
            self.conv_layers.add_module(f'bn_{i}', nn.BatchNorm2d(conv_layers[i+1]))
            self.conv_layers.add_module(f'relu_{i}', nn.ReLU())
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(conv_layers[-1], 128),
            nn.Dropout(p=0.5),  # Aggressive Dropout
            nn.ReLU(),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

def train_model(train_loader, val_loader, model, optimizer, criterion, device, max_epochs, early_stopping_patience):
    best_auc = 0.0
    patience_counter = 0

    for epoch in range(max_epochs):
        model.train()
        total_loss = 0.0
        for batch_idx, (inputs, labels) in enumerate(train_loader):
            inputs, labels = inputs.to(device), labels.to(device)
            
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels.float())
            loss.backward()
            optimizer.step()

            total_loss += loss.item()
            torch.cuda.empty_cache()  # Clear GPU cache

        avg_train_loss = total_loss / (batch_idx + 1)
        print(f'Epoch [{epoch+1}/{max_epochs}], Train Loss: {avg_train_loss:.4f}')

        # Validate
        model.eval()
        all_preds = []
        all_labels = []
        with torch.no_grad():
            for inputs, labels in val_loader:
                inputs, labels = inputs.to(device), labels.to(device)
                outputs = model(inputs)
                all_preds.extend(outputs.cpu().numpy())
                all_labels.extend(labels.cpu().numpy())

        from sklearn.metrics import roc_auc_score
        try:
            auc = roc_auc_score(all_labels, all_preds, average='weighted')
        except ValueError:
            auc = 0.0

        print(f'Validation AUC: {auc:.4f}')
        
        if auc > best_auc:
            best_auc = auc
            patience_counter = 0
            torch.save(model.state_dict(), 'model.pt')  # Save the model checkpoint
        else:
            patience_counter += 1
            if patience_counter >= early_stopping_patience:
                print("Early Stopping Activated.")
                break

    return best_auc, avg_train_loss

def main():
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    
    # Load data
    train_loader, val_loader = load_data()
    
    # Hyperparameters
    iteration = 132  # Example iteration number
    batch_size = [8, 16, 32][iteration % 3]
    lr = [0.001, 0.0001, 0.00001][iteration % 3]
    conv_layers = [64, 128, 256][:3 + (iteration % 3)]
    dropout_rate = [0.0, 0.2, 0.5][iteration % 3]
    
    # Model
    model = BirdCLEFModel(NUM_CLASSES, conv_layers).to(device)
    optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=1e-4)  # Example L2 regularization
    criterion = nn.BCELoss()
    
    # Train
    final_auc, final_train_loss = train_model(train_loader, val_loader, model, optimizer, criterion, device, MAX_EPOCHS, EARLY_STOPPING_PATIENCE)
    
    # Save metrics
    metrics = {
        "final_train_loss": final_train_loss,
        "final_auc": final_auc,
        "num_params": sum(p.numel() for p in model.parameters()),
        "epochs_trained": MAX_EPOCHS,
        "batch_size": batch_size,
        "training_samples": len(train_loader.dataset),
        "eval_samples": len(val_loader.dataset)
    }
    
    with open('metrics.json', 'w') as f:
        json.dump(metrics, f)
    
    print("METRICS: " + json.dumps(metrics))

if __name__ == "__main__":
    main()
```

### Explanation:

1. **Model Architecture**:
   - The model consists of a series of convolutional layers followed by batch normalization and ReLU activation.
   - An adaptive average pooling layer is used to flatten the output before passing it through fully connected layers.
   - Dropout is applied in the classifier layer for regularization.

2. **Training Loop**:
   - The training loop iterates over the dataset, updating weights using Adam optimizer.
   - Loss is computed using Binary Cross-Entropy (BCE) loss and backpropagation.

3. **Validation and Metrics**:
   - Validation is performed after each epoch to compute the AUC metric.
   - Early stopping is implemented based on the validation AUC to prevent overfitting.

4. **Hyperparameter Variation**:
   - The script includes a systematic variation of hyperparameters based on the iteration number, as described in the guidelines.

5. **Saving Results**:
   - Training loss and validation AUC are saved to a JSON file for further analysis.
   - Model weights are saved for reuse or submission.
