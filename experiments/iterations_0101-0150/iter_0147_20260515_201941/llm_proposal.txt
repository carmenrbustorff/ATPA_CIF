```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Constants
NUM_SPECIES = 206
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")

# Hyperparameters
max_epochs = 40
early_stopping_patience = 5

# Model Architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, num_conv_layers=3, dropout_rate=0.2):
        super(BirdCLEFModel, self).__init__()
        channels = [32, 64, 128, 256]
        layers = []
        
        for i in range(num_conv_layers):
            in_channels = 1 if i == 0 else channels[i-1]
            out_channels = channels[i]
            layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            layers.append(nn.BatchNorm2d(out_channels))
            layers.append(nn.ReLU())
        
        self.conv_layers = nn.Sequential(*layers)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 128),
            nn.Dropout(dropout_rate),
            nn.ReLU(),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data Loading
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=32,
    augment=True,
    val_split=0.2,
)

# Training Loop
def train_model(max_epochs, early_stopping_patience):
    model = BirdCLEFModel(num_species=NUM_SPECIES)
    model.to('cuda')
    
    optimizer = optim.Adam(model.parameters(), lr=0.001, weight_decay=1e-4)
    criterion = nn.BCELoss()
    
    best_auc = 0
    patience_counter = 0
    
    for epoch in range(max_epochs):
        model.train()
        running_loss = 0.0
        
        for batch_idx, (inputs, labels) in enumerate(train_loader):
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            
            try:
                optimizer.zero_grad()
                outputs = model(inputs)
                loss = criterion(outputs, labels.float())
                loss.backward()
                optimizer.step()
                
                running_loss += loss.item()
                torch.cuda.empty_cache()
            except RuntimeError as e:
                if 'out of memory' in str(e):
                    print("CUDA out of memory. Clearing cache and reducing batch size.")
                    torch.cuda.empty_cache()
                    continue
                else:
                    raise e
            
            if batch_idx % 10 == 0:  # Report loss per batch
                print(f"Epoch {epoch+1}/{max_epochs}, Batch {batch_idx}/{len(train_loader)}, Loss: {loss.item():.4f}")
        
        # Validation
        model.eval()
        all_preds = []
        all_labels = []
        
        with torch.no_grad():
            for inputs, labels in val_loader:
                inputs, labels = inputs.to('cuda'), labels.to('cuda')
                outputs = model(inputs)
                all_preds.extend(outputs.cpu().numpy())
                all_labels.extend(labels.cpu().numpy())
        
        # Calculate AUC
        from sklearn.metrics import roc_auc_score
        auc = roc_auc_score(all_labels, all_preds, average='weighted')
        print(f"Epoch {epoch+1}/{max_epochs}, Validation AUC: {auc:.4f}")
        
        # Early Stopping
        if auc > best_auc:
            best_auc = auc
            patience_counter = 0
            torch.save(model.state_dict(), 'model.pt')
        else:
            patience_counter += 1
            if patience_counter >= early_stopping_patience:
                print("Early stopping triggered.")
                break
    
    return running_loss / len(train_loader), best_auc

# Training and Evaluation
final_train_loss, final_auc = train_model(max_epochs, early_stopping_patience)

# Save Metrics
metrics = {
    "final_train_loss": final_train_loss,
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": max_epochs,
    "batch_size": 32,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

This script defines a new model architecture with varying depths and dropout rates based on the iteration number. It includes a training loop with early stopping and saves metrics and model checkpoints as required. The model uses adaptive average pooling for robust flattening and applies sigmoid activation in the forward pass to ensure outputs are in the [0, 1] range for BCE loss.
