```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Define the model architecture
class BirdClassifier(nn.Module):
    def __init__(self, num_species, num_conv_layers=3, dropout_rate=0.5):
        super(BirdClassifier, self).__init__()
        
        # Convolutional layers
        channels = [16, 32, 64, 128][:num_conv_layers]
        conv_blocks = []
        for i in range(num_conv_layers):
            in_channels = channels[i-1] if i > 0 else 1
            out_channels = channels[i]
            conv_blocks.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            conv_blocks.append(nn.BatchNorm2d(out_channels))
            conv_blocks.append(nn.ReLU())
        self.conv_layers = nn.Sequential(*conv_blocks)
        
        # Pooling and flattening
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Classifier
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 256),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(256, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Hyperparameters
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_EPOCHS = 20
BATCH_SIZE = 32
LEARNING_RATE = 0.001
DROPOUT_RATE = 0.5
NUM_CONV_LAYERS = 3

# Build dataloaders
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

# Initialize model, loss function, and optimizer
model = BirdClassifier(num_species=train_loader.dataset.num_classes)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-4)

# Move model to GPU if available
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Training loop
for epoch in range(NUM_EPOCHS):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}"):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Zero the parameter gradients
        optimizer.zero_grad()
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        
        # Backward and optimize
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
    
    # Print average training loss per epoch
    print(f"Epoch {epoch+1}, Average Training Loss: {running_loss/len(train_loader):.4f}")
    
    # Validate the model
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    
    # Calculate AUC
    from sklearn.metrics import roc_auc_score
    final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Validation AUC: {final_auc:.4f}")

# Save metrics and model checkpoint
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": NUM_EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**:
   - The model consists of a series of convolutional layers followed by batch normalization and ReLU activation.
   - An adaptive average pooling layer is used to flatten the output.
   - A fully connected layer with dropout regularization is added for classification.

2. **Hyperparameters**:
   - Number of epochs, batch size, learning rate, dropout rate, and number of convolutional layers are set based on the iteration rules provided.

3. **Data Loading**:
   - The `build_train_val_dataloaders` function from the provided data loader is used to create train and validation dataloaders.

4. **Training Loop**:
   - The model is trained for a specified number of epochs.
   - Training loss is printed at the end of each epoch.
   - Validation AUC is calculated using `roc_auc_score`.

5. **Metrics and Model Saving**:
   - Metrics are saved to `metrics.json` with keys as specified.
   - The model's state dictionary is saved to `model.pt`.

This setup allows for rapid experimentation by varying hyperparameters and observing the impact on validation AUC.
