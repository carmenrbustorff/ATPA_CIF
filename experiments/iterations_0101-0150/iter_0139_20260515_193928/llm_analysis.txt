class EarlyStopping:
    def __init__(self, patience=5):
        self.patience = patience
        self.best_auc = -np.inf
        self.wait = 0

    def step(self, auc):
        if auc > self.best_auc:
            self.best_auc = auc
            self.wait = 0
        else:
            self.wait += 1
            if self.wait >= self.patience:
                return True
        return False
