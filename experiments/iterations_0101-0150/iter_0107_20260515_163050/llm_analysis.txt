```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=4)  # Reduced batch size to 4

NUM_SPECIES = 206
epochs = 6
lr = 0.005
dropout_rate = 0.3

class BirdCLEFModel(nn.Module):
    def __init__(self, num_species):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Dropout2d(p=dropout_rate),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
            nn.Dropout2d(p=dropout_rate),
            nn.Conv2d(128, 256, 3, padding=1), nn.BatchNorm2d(256), nn.ReLU(),  # Additional layer
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(256, 128), nn.ReLU(),
            nn.Dropout(p=dropout_rate),
            nn.Linear(128, num_species)  # Use num_species dynamically
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).flatten(1)
        x = self.classifier(x)
        return torch.sigmoid(x)  # Ensure outputs are in [0, 1]

model = BirdCLEFModel(NUM_SPECIES).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

train_subset = torch.utils.data.Subset(train_loader.dataset, range(15000))
train_loader = torch.utils.data.DataLoader(train_subset, batch_size=train_loader.batch_size, shuffle=True)

for epoch in tqdm(range(epochs)):
    model.train()
    running_loss = 0.0
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        running_loss += loss.item()
    
    print(f'Epoch {epoch+1}, Train Loss: {running_loss/len(train_loader)}')

# Validation
model.eval()
all_labels = []
all_preds = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        all_labels.extend(labels.cpu().numpy())
        all_preds.extend(outputs.cpu().numpy())

from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": train_loader.batch_size,
    "training_samples": len(train_subset),
    "eval_samples": len(train_subset)
}
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + str(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```
