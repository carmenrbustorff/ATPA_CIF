```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_SPECIES = 206

# Hyperparameters based on iteration number (Example: Iteration 1)
batch_size = 16  # Iteration % 3 == 1
lr = 0.0001      # Iteration % 3 == 1
num_conv_layers = 3   # Iteration % 3 == 1
dropout_rate = 0.2    # Varying dropout rate

# Data loading
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Limit training dataset to 15000 samples for faster iteration
subset = torch.utils.data.Subset(train_loader.dataset, range(min(15000, len(train_loader.dataset))))
train_loader.dataset = subset

# Model Definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, dropout_rate=dropout_rate):
        super(BirdCLEFModel, self).__init__()
        
        # Convolutional layers
        conv_layers = []
        in_channels = 1
        out_channels = 32
        
        for _ in range(num_conv_layers):
            conv_layers.append(nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            conv_layers.append(nn.BatchNorm2d(out_channels))
            conv_layers.append(nn.ReLU())
            in_channels = out_channels
            out_channels *= 2
        
        self.conv_layers = nn.Sequential(*conv_layers)
        
        # Adaptive pooling and flattening
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        
        # Fully connected layers
        self.classifier = nn.Sequential(
            nn.Linear(in_channels, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, optimizer, and loss function
model = BirdCLEFModel()
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Move model to GPU if available
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
model.to(device)

# Training loop
epochs = 6
for epoch in range(epochs):
    model.train()
    running_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        
        # Backward pass and optimization
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
        
    print(f'Epoch [{epoch+1}/{epochs}], Loss: {running_loss/(batch_idx+1):.4f}')

# Evaluation on validation set
model.eval()
val_subset = torch.utils.data.Subset(train_loader.dataset, range(min(5000, len(train_loader.dataset))))
val_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)
val_loader.dataset = val_subset

all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

from sklearn.metrics import roc_auc_score
import numpy as np

all_preds = np.concatenate(all_preds, axis=0)
all_labels = np.concatenate(all_labels, axis=0)

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f'Validation AUC: {final_auc:.4f}')

# Save metrics to JSON
metrics = {
    "final_train_loss": running_loss/(batch_idx+1),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": len(subset.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:
1. **Imports and Setup**: The code starts with the required imports and sets up the necessary paths and constants.
2. **Hyperparameters**: Based on the iteration number, specific hyperparameters are chosen for batch size, learning rate, number of convolutional layers, and dropout rate.
3. **Data Loading**: The `get_dataloader` function is used to load the dataset. A subset of 15000 samples is taken for training due to constraints.
4. **Model Architecture**: A CNN model is defined with a flexible number of convolutional layers, batch normalization, ReLU activation, adaptive average pooling, and fully connected layers with dropout.
5. **Training Loop**: The model is trained for 6 epochs, and the loss per epoch is reported.
6. **Evaluation**: The model is evaluated on a validation subset of 5000 samples, and the AUC score is computed.
7. **Metrics and Model Saving**: Training metrics are saved to `metrics.json`, and the model's state dictionary is saved for later use.
