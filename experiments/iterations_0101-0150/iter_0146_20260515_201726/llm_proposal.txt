Here's an improved model architecture designed for the BirdCLEF+ 2026 competition (Track B). This architecture includes the necessary components like convolutional layers, batch normalization, adaptive average pooling, and a fully connected classifier with dropout for regularization.

```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Hyperparameters
NUM_CLASSES = 206
LEARNING_RATE = 0.001
BATCH_SIZE = 32
EPOCHS = 20
DROPOUT_RATE = 0.5
WEIGHT_DECAY = 1e-4

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

# Model definition
class BirdClassifier(nn.Module):
    def __init__(self, num_classes):
        super(BirdClassifier, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.MaxPool2d(2),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(64 * 32 * 27, 128), nn.ReLU(), nn.Dropout(DROPOUT_RATE),
            nn.Linear(128, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Initialize model, loss function, and optimizer
model = BirdClassifier(NUM_CLASSES).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY)

# Training loop
best_auc = 0.0
early_stopping_counter = 0
max_early_stops = 5

for epoch in range(EPOCHS):
    model.train()
    running_loss = 0.0
    
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        torch.cuda.empty_cache()  # Prevent GPU fragmentation
    
    print(f"Epoch {epoch+1}, Loss: {running_loss/len(train_loader)}")
    
    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    
    auc_score = roc_auc_score(all_labels, all_preds, average='weighted', multi_class='ovr')
    print(f"Validation AUC: {auc_score}")
    
    if auc_score > best_auc:
        best_auc = auc_score
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')  # Save the best model
    else:
        early_stopping_counter += 1
    
    if early_stopping_counter >= max_early_stops:
        print("Early stopping triggered.")
        break

# Metrics capture & saving
num_params = sum(p.numel() for p in model.parameters())
metrics = {
    "final_train_loss": running_loss/len(train_loader),
    "final_auc": auc_score,
    "num_params": num_params,
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Key Improvements and Considerations:

1. **Model Architecture**:
   - The model uses a simple yet effective architecture with convolutional layers, batch normalization, max pooling, and dropout for regularization.
   - The adaptive average pooling layer ensures robust flattening of the feature maps.

2. **Training Loop**:
   - The training loop includes moving data to CUDA, computing loss, backpropagation, and optimizer steps.
   - GPU cache is cleared after each iteration to prevent memory fragmentation.

3. **Validation and Early Stopping**:
   - AUC score is computed on the validation set after each epoch.
   - Early stopping is implemented to halt training when performance does not improve for a specified number of epochs.

4. **Metrics and Model Saving**:
   - Training loss, final AUC, number of parameters, and other relevant metrics are captured and saved in JSON format.
   - The best model weights are saved based on the highest validation AUC score.

This approach allows for rapid experimentation with different hyperparameters and architectures while ensuring that the model is well-validated and saved appropriately.
