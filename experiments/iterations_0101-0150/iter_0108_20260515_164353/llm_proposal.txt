```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Hyperparameters
epochs = 6
batch_size = 16  # Adjust based on iteration number (e.g., 8, 16, 32)
lr = 0.0001     # Adjust based on iteration number (e.g., 0.001, 0.0001, 0.00001)
dropout_rate = 0.2  # Adjust based on iteration number (e.g., 0.0, 0.2, 0.5)

# Model definition
class BirdClassifier(nn.Module):
    def __init__(self, num_species):
        super(BirdClassifier, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),  # Add another layer for depth
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(128, 256), nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(256, num_species)  # NO BatchNorm1d!
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Subset for faster iteration
subset = torch.utils.data.Subset(train_loader.dataset, range(min(15000, len(train_loader.dataset))))
train_loader = torch.utils.data.DataLoader(subset, batch_size=batch_size, shuffle=True)

# Initialize model, loss function, and optimizer
num_species = 206  # Assuming this is the number of unique species in your dataset
model = BirdClassifier(num_species).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.0001)

# Training loop
train_losses = []
for epoch in range(epochs):
    model.train()
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        print(f'Epoch {epoch+1}/{epochs}, Batch Loss: {loss.item():.4f}')
        
        torch.cuda.empty_cache()  # Prevent GPU fragmentation

# Validation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in train_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Evaluate AUC
from sklearn.metrics import roc_auc_score
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics

    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": len(subset),
    "eval_samples": len(subset)  # Using same subset for simplicity
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: 
   - The model uses 3 convolutional layers followed by batch normalization and ReLU activation.
   - An AdaptiveAvgPool2d layer is used to reduce the spatial dimensions to (1, 1), which is then flattened.
   - Two fully connected layers are used for classification with dropout applied in between.

2. **Training Loop**:
   - The model is trained for a specified number of epochs (6 in this case).
   - Losses are tracked per batch and printed out.
   - `torch.cuda.empty_cache()` is called after each batch to prevent GPU memory fragmentation.

3. **Validation**:
   - The same DataLoader is used for validation, but the model is set to evaluation mode (`model.eval()`) and gradients are disabled (`torch.no_grad()`).
   - Predictions and true labels are collected for AUC calculation.

4. **Metrics and Model Saving**:
   - The final training loss, AUC score, number of parameters, hyperparameters used, and dataset sizes are saved to a JSON file.
   - The model's state dictionary is saved to a file for later use or submission.

This architecture and training loop provide a balanced approach suitable for rapid experimentation while ensuring that key performance metrics are captured.
