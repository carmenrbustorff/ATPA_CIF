import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Define hyperparameters
iteration = 4
num_species = 206
batch_size = 16 if iteration % 3 == 1 else (8 if iteration % 3 == 0 else 32)
lr = 0.0001 if iteration % 3 == 1 else (0.001 if iteration % 3 == 0 else 0.00001)
dropout_rate = 0.3

# Data ingestion
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Define model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species, dropout_rate=dropout_rate):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(64, 128), nn.ReLU(), nn.Dropout(dropout_rate),
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_species=num_species)
model.to('cuda')

# Define loss function and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr, weight_decay=0.0001)

# Training loop
epochs = 6
training_samples = min(15000, len(train_loader.dataset))
val_samples = min(5000, len(train_loader.dataset))

subset_indices = torch.randperm(len(train_loader.dataset))[:training_samples]
train_subset = torch.utils.data.Subset(train_loader.dataset, subset_indices)
train_loader = torch.utils.data.DataLoader(train_subset, batch_size=batch_size, shuffle=True)

# Validation setup
val_subset_indices = torch.randperm(len(train_loader.dataset))[training_samples:training_samples + val_samples]
val_subset = torch.utils.data.Subset(train_loader.dataset, val_subset_indices)
val_loader = torch.utils.data.DataLoader(val_subset, batch_size=32, shuffle=False)

train_losses = []
model.train()
for epoch in range(epochs):
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch + 1}"):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        torch.cuda.empty_cache()

# Evaluate on validation set
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        outputs = model(inputs)
        
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

# Compute AUC
from sklearn.metrics import roc_auc_score

all_preds = np.vstack(all_preds)
all_labels = np.vstack(all_labels)
final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

# Save metrics to JSON
metrics = {
    "final_train_loss": np.mean(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": training_samples,

}

with open('metrics.json', 'w') as f:

print("METRICS: " + json.dumps(metrics))

# Save model checkpoint
torch.save(model.state_dict(), 'model.pt')
