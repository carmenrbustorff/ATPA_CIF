   ```python
   from sklearn.metrics import roc_auc_score

   def train_and_evaluate(model, train_loader, val_loader, epochs=20):
       criterion = nn.BCELoss()
       optimizer = torch.optim.Adam(model.parameters(), lr=lr)

       best_auc = 0.0
       patience = 5
       counter = 0

       for epoch in range(epochs):
           model.train()
           running_loss = 0.0
           
           for inputs, labels in train_loader:
               inputs, labels = inputs.to('cuda'), labels.to('cuda')
               
               optimizer.zero_grad()
               outputs = model(inputs)
               loss = criterion(outputs, labels)
               loss.backward()
               optimizer.step()
               
               running_loss += loss.item()

           # Validation
           model.eval()
           all_preds = []
           all_labels = []

           with torch.no_grad():
               for inputs, labels in val_loader:
                   inputs, labels = inputs.to('cuda'), labels.to('cuda')
                   outputs = model(inputs)
                   all_preds.append(outputs.cpu().numpy())
                   all_labels.append(labels.cpu().numpy())

           # Flatten lists
           all_preds = np.concatenate(all_preds)
           all_labels = np.concatenate(all_labels)

           val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
           print(f"Epoch {epoch+1}, Train Loss: {running_loss/len(train_loader)}, Val AUC: {val_auc}")

           # Early stopping
           if val_auc > best_auc:
               best_auc = val_auc
               counter = 0
               torch.save(model.state_dict(), 'model.pt')  # Save the best model
           else:
               counter += 1
               if counter >= patience:
                   print("Early stopping triggered.")
                   break

       return running_loss/len(train_loader), best_auc

   final_train_loss, final_auc = train_and_evaluate(model, train_loader, val_loader)
   ```
