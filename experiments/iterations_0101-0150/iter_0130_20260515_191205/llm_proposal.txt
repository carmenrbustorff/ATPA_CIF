```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm

# Define the model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species=206):
        super(BirdCLEFModel, self).__init__()
        # Convolutional layers
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, 3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, 3, padding=1), nn.BatchNorm2d(128), nn.ReLU()
        )
        # Adaptive pooling
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        # Classifier layers
        self.classifier = nn.Sequential(
            nn.Linear(128, 64), nn.ReLU(), nn.Dropout(0.5),
            nn.Linear(64, num_species)  # NO BatchNorm1d!
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Hyperparameters
NUM_EPOCHS = 20
BATCH_SIZE = 32
LEARNING_RATE = 0.0001
DROPOUT_RATE = 0.5

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2
)

# Model initialization
model = BirdCLEFModel(num_species=train_loader.dataset.num_classes)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-4)

# Move model to GPU if available
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Training loop
for epoch in range(NUM_EPOCHS):
    model.train()
    running_loss = 0.0
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        
        # Backward pass and optimization
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        running_loss += loss.item()
        
        # Clear cache to prevent GPU fragmentation
        torch.cuda.empty_cache()
        
        if batch_idx % 10 == 9:  # Print every 10 batches
            print(f'Epoch [{epoch+1}/{NUM_EPOCHS}], Step [{batch_idx+1}/{len(train_loader)}], Loss: {running_loss/10:.4f}')
            running_loss = 0.0

# Evaluation on validation set
model.eval()
all_preds, all_labels = [], []
with torch.no_grad():
    for inputs, labels in val_loader:
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.append(outputs.cpu().numpy())
        all_labels.append(labels.cpu().numpy())

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')

print(f"Final Validation AUC: {final_auc:.4f}")

# Save metrics and model checkpoint
metrics = {
    "final_train_loss": running_loss / len(train_loader),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": NUM_EPOCHS,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print(f"METRICS: {json.dumps(metrics)}")

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Model Architecture**: The model is defined with 4 convolutional layers followed by adaptive pooling and a classifier with dropout for regularization.
2. **Hyperparameters**: The hyperparameters are set to a moderate level of complexity, suitable for rapid experimentation.
3. **Data Loading**: The training and validation datasets are loaded using the `build_train_val_dataloaders` function.
4. **Training Loop**: The model is trained for 20 epochs with BCELoss and Adam optimizer. Loss is printed every 10 batches.
5. **Evaluation**: After training, the model is evaluated on the validation set to compute the AUC score.
6. **Metrics and Checkpoint Saving**: The final metrics are saved to a JSON file, and the model checkpoint is saved for later use.

This setup allows for rapid experimentation with different hyperparameters and architectures while ensuring that results are reproducible and loggable.
