```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Define the model architecture
class BirdCLEFModel(nn.Module):
    def __init__(self, num_species):
        super(BirdCLEFModel, self).__init__()
        # Number of convolutional layers based on iteration number
        num_conv_layers = 3  # Change this based on iteration logic
        channels = [16, 32, 64]  # Example channel sizes
        
        self.conv_layers = nn.Sequential()
        for i in range(num_conv_layers):
            in_channels = channels[i-1] if i > 0 else 1
            out_channels = channels[i]
            self.conv_layers.add_module(f'conv_{i}', nn.Conv2d(in_channels, out_channels, 3, padding=1))
            self.conv_layers.add_module(f'bn_{i}', nn.BatchNorm2d(out_channels))
            self.conv_layers.add_module(f'relu_{i}', nn.ReLU())
        
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(channels[-1], 128),
            nn.ReLU(),
            nn.Dropout(0.5),  # Regularization
            nn.Linear(128, num_species)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

# Hyperparameters
NUM_SPECIES = 206
BATCH_SIZE = 32  # Adjust based on iteration logic
LEARNING_RATE = 0.0001  # Adjust based on iteration logic
EPOCHS = 20
EARLY_STOPPING_PATIENCE = 5

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2
)

# Model setup
model = BirdCLEFModel(NUM_SPECIES).to('cuda')
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=1e-4)

# Training loop
best_val_auc = 0
early_stopping_counter = 0

for epoch in range(EPOCHS):
    model.train()
    running_loss = 0.0
    
    for batch_idx, (inputs, labels) in enumerate(train_loader):
        inputs, labels = inputs.to('cuda'), labels.to('cuda')
        
        try:
            optimizer.zero_grad()
            outputs = model(inputs)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()
            
            running_loss += loss.item()
            torch.cuda.empty_cache()  # Prevent GPU fragmentation
            
        except RuntimeError as e:
            if 'out of memory' in str(e):
                print(f"OOM caught. Reducing batch size from {BATCH_SIZE} to {BATCH_SIZE // 2}")
                BATCH_SIZE //= 2
                break
        
        if batch_idx % 10 == 9:  # Report every 10 batches
            print(f'Epoch [{epoch+1}/{EPOCHS}], Batch [{batch_idx+1}/{len(train_loader)}], Loss: {loss.item():.4f}')
    
    # Validation loop
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to('cuda'), labels.to('cuda')
            outputs = model(inputs)
            all_preds.extend(outputs.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
    
    # Compute AUC
    all_preds = np.concatenate(all_preds)
    all_labels = np.concatenate(all_labels)
    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f'Validation AUC: {val_auc:.4f}')
    
    # Early stopping check
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        early_stopping_counter = 0
        torch.save(model.state_dict(), 'model.pt')  # Save the model
    else:
        early_stopping_counter += 1
        if early_stopping_counter >= EARLY_STOPPING_PATIENCE:
            print("Early stopping triggered.")
            break

# Save metrics
metrics = {
    "final_train_loss": running_loss / len(train_loader),
    "final_auc": best_val_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epoch + 1,
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))
```

### Explanation:

1. **Model Architecture**:
   - The model consists of a series of convolutional layers followed by batch normalization and ReLU activation.
   - An adaptive average pooling layer is used to flatten the output.
   - A fully connected layer with dropout for regularization and another linear layer for output.

2. **Hyperparameters**:
   - Adjusted based on the iteration logic provided (e.g., number of convolutional layers, learning rate).

3. **Data Loading**:
   - Uses `build_train_val_dataloaders` to create training and validation datasets.

4. **Training Loop**:
   - Moves model and data to CUDA.
   - Handles out-of-memory exceptions by reducing the batch size.
   - Reports loss every 10 batches.

5. **Validation Loop**:
   - Computes AUC using `roc_auc_score`.
   - Implements early stopping based on validation AUC improvement.

6. **Metrics and Model Saving**:
   - Saves metrics to `metrics.json` and prints them for logging.
   - Saves the best model weights to `model.pt`.
