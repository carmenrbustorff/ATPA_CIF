  import torch.nn as nn
  import torch.optim as optim
  from sklearn.metrics import roc_auc_score

  # Define hyperparameters
  epochs = 6
  num_species = 206
  dropout_rate = 0.5

  # Model definition
  class BirdClassifier(nn.Module):
      def __init__(self, num_species=num_species, dropout_rate=dropout_rate):
          super(BirdClassifier, self).__init__()
          self.conv_layers = nn.Sequential(
              nn.Conv2d(1, 32, 3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
          )
          self.pool = nn.AdaptiveAvgPool2d((1, 1))
          self.classifier = nn.Sequential(
              nn.Linear(32, 64), nn.ReLU(), nn.Dropout(dropout_rate),
              nn.Linear(64, num_species)
          )

      def forward(self, x):
          x = self.conv_layers(x)
          x = self.pool(x).view(x.size(0), -1)
          return torch.sigmoid(self.classifier(x))

  model = BirdClassifier().to('cuda')
  criterion = nn.BCELoss()
  optimizer = optim.Adam(model.parameters(), lr=lr)

  # Training loop
  train_subset = torch.utils.data.Subset(train_loader.dataset, range(min(15000, len(train_loader.dataset))))
  train_loader = torch.utils.data.DataLoader(train_subset, batch_size=batch_size, shuffle=True)

  for epoch in range(epochs):
      model.train()
      running_loss = 0.0
      for inputs, labels in train_loader:
          inputs, labels = inputs.to('cuda'), labels.to('cuda')
          
          optimizer.zero_grad()
          outputs = model(inputs)
          loss = criterion(outputs, labels)
          loss.backward()
          optimizer.step()
          
          running_loss += loss.item()
          torch.cuda.empty_cache()  # Prevent GPU fragmentation

      print(f'Epoch {epoch + 1}, Training Loss: {running_loss / len(train_loader):.4f}')
