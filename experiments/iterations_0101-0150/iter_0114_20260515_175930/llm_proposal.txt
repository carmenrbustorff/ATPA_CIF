```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from data_loader import build_train_val_dataloaders
import json
from tqdm import tqdm
from sklearn.metrics import roc_auc_score

# Hyperparameters
NUM_EPOCHS = 20
BATCH_SIZE = 32
LEARNING_RATE = 0.0001
NUM_CLASSES = 206
DROPOUT_RATE = 0.5
WEIGHT_DECAY = 1e-4

# Data loading
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
train_loader, val_loader = build_train_val_dataloaders(
    metadata_csv=METADATA_FILE,
    audio_dir=DATA_DIR,
    batch_size=BATCH_SIZE,
    augment=True,
    val_split=0.2,
)

# Model definition
class BirdCLEFModel(nn.Module):
    def __init__(self, num_classes):
        super(BirdCLEFModel, self).__init__()
        self.conv_layers = nn.Sequential(
            nn.Conv2d(1, 32, kernel_size=3, padding=1), nn.BatchNorm2d(32), nn.ReLU(),
            nn.Conv2d(32, 64, kernel_size=3, padding=1), nn.BatchNorm2d(64), nn.ReLU(),
            nn.Conv2d(64, 128, kernel_size=3, padding=1), nn.BatchNorm2d(128), nn.ReLU(),
        )
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(128, 64), nn.ReLU(), nn.Dropout(DROPOUT_RATE),
            nn.Linear(64, num_classes)
        )

    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdCLEFModel(num_classes=NUM_CLASSES)

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=LEARNING_RATE, weight_decay=WEIGHT_DECAY)

# Move model to GPU if available
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Early stopping parameters
early_stopping_patience = 5
best_val_auc = 0.0
epochs_without_improvement = 0

# Training loop
for epoch in range(NUM_EPOCHS):
    model.train()
    running_loss = 0.0
    for inputs, labels in tqdm(train_loader, desc=f"Epoch {epoch+1}"):
        inputs, labels = inputs.to(device), labels.to(device)
        
        # Zero the parameter gradients
        optimizer.zero_grad()
        
        # Forward pass
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        
        # Backward pass and optimization
        loss.backward()
        optimizer.step()
        
        # Update running loss
        running_loss += loss.item()
        
        # Clear cache to prevent GPU fragmentation
        torch.cuda.empty_cache()

    epoch_loss = running_loss / len(train_loader)
    print(f"Epoch {epoch+1}, Loss: {epoch_loss:.4f}")

    # Validation
    model.eval()
    all_preds = []
    all_labels = []
    
    with torch.no_grad():
        for inputs, labels in val_loader:
            inputs, labels = inputs.to(device), labels.to(device)
            outputs = model(inputs)
            all_preds.append(outputs.cpu().numpy())
            all_labels.append(labels.cpu().numpy())

    all_preds = np.concatenate(all_preds)
    all_labels = np.concatenate(all_labels)

    val_auc = roc_auc_score(all_labels, all_preds, average='weighted')
    print(f"Validation AUC: {val_auc:.4f}")

    # Early stopping
    if val_auc > best_val_auc:
        best_val_auc = val_auc
        epochs_without_improvement = 0
        torch.save(model.state_dict(), 'model.pt')  # Save the best model
    else:
        epochs_without_improvement += 1

    if epochs_without_improvement >= early_stopping_patience:
        print(f"Early stopping triggered after {early_stopping_patience} epochs without improvement.")
        break

# Metrics and final AUC computation
final_auc = val_auc
num_params = sum(p.numel() for p in model.parameters())
metrics = {
    "final_train_loss": epoch_loss,
    "final_auc": final_auc,
    "num_params": num_params,
    "epochs_trained": NUM_EPOCHS - epochs_without_improvement,  # Adjust for early stopping
    "batch_size": BATCH_SIZE,
    "training_samples": len(train_loader.dataset),
    "eval_samples": len(val_loader.dataset)
}

# Save metrics to JSON
with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

# Print metrics for log parsing
print("METRICS: " + json.dumps(metrics))
```
