```python
import os
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch.optim as optim
from dataset import get_dataloader
import json
from tqdm import tqdm

# Constants for this iteration
DATA_DIR = "/mnt/disks/data/birdclef"
METADATA_FILE = os.path.join(DATA_DIR, "train.csv")
NUM_SPECIES = 206
ITERATION = 10  # Current iteration number for systematic hyperparameter exploration

# Hyperparameters based on iteration number
if ITERATION % 3 == 0:
    batch_size = 8
    lr = 0.001
    num_conv_layers = 2
    dropout_rate = 0.5
elif ITERATION % 3 == 1:
    batch_size = 16
    lr = 0.0001
    num_conv_layers = 3
    dropout_rate = 0.2
else:
    batch_size = 32
    lr = 0.00001
    num_conv_layers = 4
    dropout_rate = 0.0

# Data loading
train_loader = get_dataloader(DATA_DIR, METADATA_FILE, batch_size=batch_size)

# Limit training and validation datasets for faster iteration
training_samples = 15000
eval_samples = 5000
subset_indices = torch.randperm(len(train_loader.dataset))[:training_samples]
val_subset_indices = torch.randperm(len(train_loader.dataset))[training_samples:training_samples + eval_samples]

train_subset = torch.utils.data.Subset(train_loader.dataset, subset_indices)
val_subset = torch.utils.data.Subset(train_loader.dataset, val_subset_indices)

train_loader.dataset.set_mode('train')
train_loader = torch.utils.data.DataLoader(train_subset, batch_size=batch_size, shuffle=True)

val_loader = torch.utils.data.DataLoader(val_subset, batch_size=batch_size, shuffle=False)

# Model definition
class BirdClassifier(nn.Module):
    def __init__(self, num_species=NUM_SPECIES, num_conv_layers=num_conv_layers):
        super(BirdClassifier, self).__init__()
        self.conv_layers = nn.Sequential()
        for i in range(num_conv_layers):
            in_channels = 32 if i > 0 else 1
            out_channels = 32 * (i + 1)
            self.conv_layers.add_module(f'conv_{i+1}', nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1))
            self.conv_layers.add_module(f'batchnorm_{i+1}', nn.BatchNorm2d(out_channels))
            self.conv_layers.add_module(f'relu_{i+1}', nn.ReLU())
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.classifier = nn.Sequential(
            nn.Linear(32 * num_conv_layers, 128),
            nn.ReLU(),
            nn.Dropout(dropout_rate),
            nn.Linear(128, num_species)
        )
    
    def forward(self, x):
        x = self.conv_layers(x)
        x = self.pool(x).view(x.size(0), -1)
        return torch.sigmoid(self.classifier(x))

model = BirdClassifier(num_species=NUM_SPECIES, num_conv_layers=num_conv_layers)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
model.to(device)

# Loss and optimizer
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=lr)

# Training loop
epochs = 6
train_losses = []

for epoch in range(epochs):
    model.train()
    for inputs, labels in tqdm(train_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, labels.float())
        loss.backward()
        optimizer.step()
        
        train_losses.append(loss.item())
        torch.cuda.empty_cache()  # Prevent GPU fragmentation

    print(f"Epoch {epoch+1}, Loss: {sum(train_losses) / len(train_losses):.4f}")

# Evaluation
model.eval()
all_preds = []
all_labels = []

with torch.no_grad():
    for inputs, labels in tqdm(val_loader):
        inputs, labels = inputs.to(device), labels.to(device)
        outputs = model(inputs)
        all_preds.extend(outputs.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())

# Calculate AUC
from sklearn.metrics import roc_auc_score

final_auc = roc_auc_score(all_labels, all_preds, average='weighted')
print(f"Final AUC: {final_auc:.4f}")

# Save metrics and model
metrics = {
    "final_train_loss": sum(train_losses) / len(train_losses),
    "final_auc": final_auc,
    "num_params": sum(p.numel() for p in model.parameters()),
    "epochs_trained": epochs,
    "batch_size": batch_size,
    "training_samples": training_samples,
    "eval_samples": eval_samples
}

with open('metrics.json', 'w') as f:
    json.dump(metrics, f)

print("METRICS: " + json.dumps(metrics))

torch.save(model.state_dict(), 'model.pt')
```

### Explanation:

1. **Imports and Hyperparameters**: The script starts with the required imports and sets up hyperparameters based on the iteration number using systematic exploration rules.

2. **Data Loading**: The data loader is set up, and subsets are created for training and validation to limit the dataset size for faster iterations.

3. **Model Architecture**: A `BirdClassifier` class is defined with a configurable number of convolutional layers. The architecture follows the specified pattern: Conv → BatchNorm → ReLU → Pool → Flatten → Linear.

4. **Training Loop**: The model is trained using Adam optimizer and BCELoss. Losses are tracked per batch, and GPU memory is managed by clearing caches after each batch.

5. **Evaluation**: The model is evaluated on the validation set, and AUC is calculated using `roc_auc_score`.

6. **Metrics and Model Saving**: Metrics are saved to a JSON file, and the model's state dictionary is saved for later use.

This setup prioritizes rapid experimentation by limiting dataset sizes and focusing on small-scale iterations.
